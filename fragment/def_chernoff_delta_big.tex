\begin {lemma} \lemlab {chernoff:delta:big}Let $X_1, \ldots , X_n$ be $n$ independent Bernoulli trials, where \begin {math} \Prob { X_i = 1 } =p_i, \end {math} and \begin {math} \Prob { X_i = 0 } =1- p_i, \end {math} for $i=1,\ldots , n$. Let $X = \sum _{i=1}^{b} X_i$, and $\mu = \Ex {\bigl . X} = \sum _i p_i$. For $\delta > 2e-1$, we have $\Prob { \bigl . X > (1+\delta )\mu } < 2^{-\mu (1 + \delta )}$. \end {lemma}
