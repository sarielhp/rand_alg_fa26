\begin {theorem}[Chebyshev's inequality] \thmlab {Chebyshev:inequality}Let $X$ be a real random variable, with $\mu _X = \Ex {X}$, and $\sigma _X = \sqrt {\Var {X}}$. Then, for any $t > 0$, we have \begin {equation*} \Prob {\bigl . \cardin {X - \mu _X} \geq t \sigma _X} \leq \frac {1}{t^2}. \end {equation*} \end {theorem}
