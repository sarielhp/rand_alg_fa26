\begin {theorem} \thmlab {k:wise:independent}let $p$ be a prime number, and pick independently and uniformly $k$ values $b_0. b_1, \ldots , b_{k-1} \in \ZZ _p$, and let $g(x) = \sum _{i=0}^{k-1} b_i x^i \bmod p$. Then the random variables \begin {equation*} Y_0 = g(0), \ldots , Y_{p-1} = g(p-1). \end {equation*} are uniformly distributed in $\ZZ _p$ and are $k$-wise independent. \end {theorem}
