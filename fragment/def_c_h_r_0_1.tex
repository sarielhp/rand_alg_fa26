\begin {corollary} \corlab {Chernoff:0:1}Let $X_1, \ldots , X_n$ be $n$ independent \emph {coin flips}, such that $\Prob {X_i =0} =\Prob {X_i=1} = \frac {1}{2}$, for $i=1,\ldots , n$. Let $Y = \sum _{i=1}^n X_i$. Then, for any $\Delta >0$, we have \begin {math} \Prob {\cardin {Y - {n}/{2}} \geq \Delta }\leq 2\exp \pth {-2\Delta ^2/n}. \end {math} \end {corollary}
