\begin {lemma} \lemlab {chernoff:delta:gap}Let $X_1, \ldots , X_n$ be $n$ independent random variables, where \begin {math} \Prob { X_i = 1 } =p_i, \end {math} and \begin {math} \Prob { X_i = 0 } =1- p_i, \end {math} for $i=1,\ldots , n$. Let $X = \sum _{i=1}^{b} X_i$, and $\mu = \Ex {\bigl . X} = \sum _i p_i$. For $\delta \in (0,6)$, we have \begin {equation*} \Prob { \bigl . X > (1+\delta )\mu } < \exp \pth {-\mu \delta ^2 / 5}, \end {equation*} \end {lemma}
