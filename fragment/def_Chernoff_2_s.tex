\begin{lemma}
    \lemlab{Chernoff:2:s}
    %
    Let $X_1, \ldots, X_n \in \{0,1\}$ be $n$ independent random
    variables, with
    \begin{math}
        p_i = \Prob{ \bigl. X_i = 1 },
    \end{math}
    for all $i$.  For $X = \sum_{i=1}^{n} X_i$, and
    $\mu = \Ex{\bigl. X} = \sum_i p_i$, we have that
    \begin{math}
        \Prob{\bigl. X < (1-\delta)\mu } < \exp\pth{-\mu\delta^2/2}.
    \end{math}
\end{lemma}
