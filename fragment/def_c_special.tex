\begin {theorem} \thmlab {Chernoff:special}Let $X_1, \ldots , X_n$ be $n$ independent random variables, such that $\Prob {X_i =1} =\Prob {X_i=-1} = \frac {1}{2}$, for $i=1,\ldots , n$. Let $Y = \sum _{i=1}^n X_i$. Then, for any $\Delta >0$, we have \begin {align*} \Prob {\Bigl . Y \geq \Delta } \leq \exp \pth {-\Delta ^2/2n}. \end {align*} \end {theorem}
