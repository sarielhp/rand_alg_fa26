\begin{lemma}
    \lemlab{approximation:basic}%
    %
    For any positive integer $n$, we have:
    \begin{compactenumi}
        \item $(1+1/n)^n \leq e$.

        \item $(1-1/n)^{n-1} \geq e^{-1}$.

        \item $n! \geq (n/e)^n$.

        \item For any $k \leq n$, we have:
        \begin{math}
            \ds%
            \pth[]{\frac{n}{k}}^k \leq \binom{n}{k}%
            \leq%
            \pth[]{\frac{n e}{k}}^k.
        \end{math}
    \end{compactenumi}
\end{lemma}
