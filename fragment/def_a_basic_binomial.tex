\begin {lemma} \lemlab {a:basic:binomial}For any positive integer $n$, the following hold. \smallskip \begin {compactenumi} \item $(1+1/n)^n \leq e$. \par \smallskip \item $(1-1/n)^{n-1} \geq e^{-1}$. \par \smallskip \item $n! \geq (n/e)^n$. \par \smallskip \item For any $k \leq n$, we have $\;\;\displaystyle \pth {\frac {n}{k}}^k \leq \binom {n}{k} \leq \pth {\frac {n e}{k}}^k$. \end {compactenumi} \par \index {binomial!estimates} \end {lemma}
