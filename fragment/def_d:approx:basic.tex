\begin {lemma} \lemlab {approximation:basic}For any positive integer $n$, we have: \begin {compactenumi} \medskip \item $1+x \leq e^x$ \;\;and\;\; $1-x \leq e^{-x}$, for all $x$. \par \medskip \item $(1+1/n)^n \LEQ e \LEQ \bigl (1+1/n\bigr )^{n+1}$. \par \medskip \item $(1-1/n)^n \leq \frac {1}{e} \leq (1-1/n)^{n-1} $. \par \medskip \item $(n/e)^n \LEQ n! \LEQ (n+1)^{n+1} / e^n$. \par \item For any $k \leq n$, we have: \begin {math} \ds \pth []{\frac {n}{k}}^k \LEQ \binom {n}{k}\LEQ \pth []{\frac {n e}{k}}^k. \end {math} \end {compactenumi} \end {lemma}
