\begin {lemma} \lemlab {approximation:basic}For any positive integer $n$, we have: \begin {compactenumi} \item $1+x \leq e^x$ and $1-x \leq e^{-x}$, for all $x$. \par \item $(1+1/n)^n \leq e \leq \bigl (1+1/n\bigr )^{n+1}$. \par \item $(1-1/n)^n \leq \frac {1}{e} \leq (1-1/n)^{n-1} $. \par \item $(n/e)^n \leq n! \leq (n+1)^{n+1} / e^n$. \par \item For any $k \leq n$, we have: \begin {math} \ds \pth []{\frac {n}{k}}^k \leq \binom {n}{k}\leq \pth []{\frac {n e}{k}}^k. \end {math} \end {compactenumi} \end {lemma}
