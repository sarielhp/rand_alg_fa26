\section{Martingales}

\subsection{Preliminaries}


Let $X$ and $Y$ be two random variables. Let
$\rho(x,y) = \Prob{(X = x) \cap (Y = y) }$. Observe that
\begin{align*}
  \ProbCond{X = x}{ Y = y}%
  =%
  \frac{\rho(x, y)}{\Prob{Y = y}}%
  =%
  \frac{\rho(x,y)}{\sum_{z} \rho(z, y)}
\end{align*}
and
\begin{math}
    \ds \Ex{ X \sep{ Y = y}}%
    =%
    \sum_{x} x \Prob{ X = x \sep{Y = y}} = \frac{\sum_x {} x \rho(x,
       y)}{\sum_z \rho(z, y)} = \frac{\sum_x {} x \rho(x,
       y)}{\Prob{Y=y}}.
\end{math}

The \emph{conditional expectation} \index{conditional expectation} of
$X$ given $Y$, is the random variable $\Ex{X \sep{Y}}$ is the random
variable $f(y ) =\Ex{X \sep{Y = y}}$.

\medskip%
\noindent%
As a reminder, for any two random variables $X$ and $Y$, we have
\begin{compactenumI}
    \smallskip%
    \item \itemlab{c:e:item} \lemref{conditional:expectation}:
    \begin{math}
        \Ex{ \bigl. \ExCond{X}{ Y} } = \Ex{ \bigl. X}.
    \end{math}

    \smallskip
    \item
    \lemref{expectations:product}:
    \begin{math}
        \Ex{\bigl.Y \cdot \ExCond{X }{Y}} = \Ex{\bigl. X Y}.
    \end{math}
\end{compactenumI}


\subsection{Martingales}

Intuitively, martingales are a sequence of random variables describing
a process, where the only thing that matters at the beginning of the
$i$\th step is where the process was in the end of the $(i-1)$\th
step. That is, it does not matter how the process arrived to a certain
state, only that it is currently at this state.

\begin{defn}
    A sequence of random variables $X_0, X_1, \ldots, $ is said to be
    a \emphi{martingale sequence} if for all $i > 0$, we have
    $\ExCond{X_i }{X_0, \ldots, X_{i-1}} = X_{i-1}$.
\end{defn}

In particular, note that for a martingale, we have
$\ExCond{X_i }{X_0, \ldots, X_{i-1}} = \ExCond{X_i }{X_{i-1}} =
X_{i-1}$.


\begin{lemma}
    Let $X_0, X_1, \ldots,$ be a martingale sequence. Then, for all $i
    \geq 0$, we have $\Ex{ \bigl. X_i } = \Ex{ \bigl. X_0 }$.
\end{lemma}

\begin{proof}
    By \itemref{c:e:item}, and the martingale property, we have
    \begin{equation*}
        \Ex{  X_i }
        =%
        \Ex{ \bigl.  \ExCond{X_i}{ X_{i-1}} }
        =%
        \Ex{  X_{i-1} } %
        =%
        \Ex{  X_{i-2} } %
        =%
        \cdots
        =
        \Ex{ X_{0} }. %
    \end{equation*}

\end{proof}

\subsubsection{Examples of martingales}

\begin{example}
    Consider the sum of money after participating in a sequence of
    fair bets. That is, let $X_i$ be the amount of money a gambler has
    after playing $i$ rounds. In each round it either gains one
    dollar, or loses one dollar (with equal probability). Clearly, we
    have
    \begin{equation*}
        \ExCond{ X_i}{ X_0,\ldots, X_{i-1}}%
        =%
        \ExCond{X_i}{ X_{i-1}}%
        =%
        X_{i-1}
        +
        \frac{1}{2} \cdot (+1)
        +
        \frac{1}{2} \cdot (-1)
        =%
        X_{i-1}.
    \end{equation*}
\end{example}

\begin{example}
    Let $Y_i = X_i^2 - i$, where $X_i$ is as defined in the above
    example. We claim that $Y_0, Y_1, \ldots$ is a martingale. Let us
    verify that this is true. Given $Y_{i-1}$, we have $Y_{i-1} =
    X_{i-1}^2 - (i-1)$. We have that
    \begin{align*}
        \Ex{Y_{i} \sep{ Y_{i-1}}}%
        &=%
        \Ex{X_i^2 - i \sep{ X_{i-1}^2 - (i-1) }}%
        =%
        \frac{1}{2} \pth{ \pth[]{X_{i-1}+1}^2 - i) } + \frac{1}{2}
        \pth{ \pth{X_{i-1}-1}^2 - i }\\
        &=%
        X_{i-1}^2 +1 - i%
        =%
        X_{i-1}^2 - (i-1)%
        =%
        Y_{i-1},
    \end{align*}
    which implies that indeed it is a martingale.
\end{example}


\begin{example}
    Let $U$ be a urn with $b$ black balls, and $w$ white balls. We
    repeatedly select a ball and replace it by $c$ balls having the
    same color. Let $X_i$ be the fraction of black balls after the
    first $i$ trials.  We claim that the sequence $X_0, X_1, \ldots$
    is a martingale.

    Indeed, let $n_i = b + w + i(c-1)$ be the number of balls in the
    urn after the $i$\th trial. Clearly,
    \begin{align*}
        \Ex{ X_i \sep{X_{i-1}, \ldots, X_0}}%
        &=%
        X_{i-1} \cdot \frac{(c-1) + X_{i-1}n_{i-1}}{n_{i}} + (1 -
        X_{i-1}) \cdot \frac{X_{i-1}n_{i-1}}{n_{i}}%
        \\&%
        =%
        \frac{X_{i-1}(c-1) + X_{i-1}n_{i-1}}{n_i} = X_{i-1} \frac{c-1
           + n_{i-1}}{n_i} = X_{i-1} \frac{n_{i}}{n_i} = X_{i-1}.
    \end{align*}
\end{example}


\begin{example}
    Let $\G$ be a random graph on the vertex set
    $V = \brc{1,\ldots, n}$ obtained by independently choosing to
    include each possible edge with probability $p$. The underlying
    probability space over \emphi{random graphs} is denoted by
    $\mathbf{G}_{n,p}$.  Arbitrarily label the $m = n (n-1)/2$
    possible edges with the sequence $1 ,\ldots, m$. For
    $ 1 \leq j \leq m$, define the indicator random variable $I_j$,
    which takes values $1$ if the edge $j$ is present in $\G$, and has
    value $0$ otherwise. These indicator variables are independent and
    each takes value $1$ with probability $p$.

    Consider any real valued function $f$ defined over the space of
    all graphs, e.g., the clique number, which is defined as being the
    size of the largest complete subgraph. The \emphic{edge exposure
       martingale}{martingale!edge exposure} is the sequence of random
    variables $X_0, \ldots, X_m$ such that
    \begin{align*}
      X_i = \ExCond{ f(\G)}{I_1,\ldots, I_i},
    \end{align*}
    while $X_0 = \Ex{f(G)}$ and $X_m = f(G)$. This sequence of random
    variable begin a martingale follows immediately from a theorem
    that would be described in the next lecture.

    One can define similarly a \emphic{vertex exposure
       martingale}{martingale!vertex exposure}, where the graph $G_i$
    is the graph induced on the first $i$ vertices of the random graph
    $\G$.
\end{example}


\begin{example}[The sheep of Mabinogion]
    The following is taken from medieval Welsh manuscript based on
    Celtic mythology:
    \begin{quote}
        ``And he came towards a valley, through which ran a river; and
        the borders of the valley were wooded, and on each side of the
        river were level meadows. And on one side of the river he saw
        a flock of white sheep, and on the other a flock of black
        sheep. And whenever one of the white sheep bleated, one of the
        black sheep would cross over and become white; and when one of
        the black sheep bleated, one of the white sheep would cross
        over and become black.''  -- \emph{\si{Peredur the son of
              Evrawk}}, from the \emph{\si{Mabinogion}}.
    \end{quote}

    More concretely, we start at time $0$ with $w_0$ white sheep, and
    $b_0$ black sheep. At every iteration, a random sheep is picked,
    it bleats, and a sheep of the other color turns to this color. the
    game stops as soon as all the sheep have the same color. No sheep
    dies or get born during the game.  Let $X_i$ be the expected
    number of black sheep in the end of the game, after the $i$\th
    iteration. For reasons that we would see later on, this
    sequence is a martingale.

    The original question is somewhat more interesting -- if we are
    allowed to take a way sheep in the end of each iteration, what is
    the optimal strategy to maximize $X_i$?
\end{example}

\subsubsection{Azuma's inequality}

A sequence of random variables $X_0, X_1, \ldots$ has \emphi{bounded
   differences} if $\cardin{X_i - X_{i-1}} \leq \Delta$, for some
$\Delta$.

\begin{theorem}[Azuma's Inequality.]
    Let $ X_0, \ldots, X_m$ be a martingale with $X_0 =0$, and
    \begin{equation*}
        |X_{i+1}-X_i| \leq 1, \qquad\text{ for }\quad i=0, \ldots, m-1.
    \end{equation*}
    For any $\lambda > 0$, we have
    \begin{math}\ds
        \Bigg.  \Prob{X_m > \lambda \sqrt{m}} <
        \exp\pth{-\lambda^2/2}.
    \end{math}
\end{theorem}
\begin{proof}
    Let $\alpha = \lambda / \sqrt{m}$. Let $Y_i = X_i - X_{i-1}$, so
    that $|Y_i| \leq 1$ and
    $\ExCond{ Y_i }{ X_0, \ldots, X_{i-1}} = 0$.

    We are interested in bounding $\Ex{e^{\alpha Y_i}
       \sep{ X_0, \ldots, X_{i-1}}}$. Note that, for $-1
    \leq x \leq 1$, we have
    \begin{equation*}
        f(x)
        =%
        e^{\alpha
           x} \leq h(x) = \frac{e^{\alpha} + e^{-\alpha}}{2} +
        \frac{e^{\alpha} - e^{-\alpha}}{2} x,
    \end{equation*}
    as $f(x) = e^{\alpha x}$ is a convex function,
    $h(-1) = e^{-\alpha} = f(-1)$, $h(1) = e^{\alpha} = f(+1)$, and
    $h(x)$ is a linear function.  Thus,
    \begin{align*}
      \allowdisplaybreaks%
      \Ex{ e^{\alpha Y_i} \sep{ X_0, \ldots, X_{i-1}}}
      &\leq %
        \Ex{ h( Y_i) \sep{ X_0, \ldots, X_{i-1}}} = h \pth{ \Ex{ Y_i
        \sep{ X_0, \ldots, X_{i-1}}}}%
      \\
      &%
        =%
        h\pth{\bigl.  0  }%
        =%
        \frac{e^{\alpha} +e^{-\alpha}}{2}%
      \\&%
      =%
      \frac{(1+ \alpha + \frac{\alpha^2}{2!} + \frac{\alpha^3}{3!}+
      \cdots) + (1 -\alpha + \frac{\alpha^2}{2!} -
      \frac{\alpha^3}{3!}+ \cdots)}{2}%
      \\ &%
           =%
           1 + \frac{\alpha^2}{2} + \frac{\alpha^4}{4!} +
           \frac{\alpha^6}{6!} + \cdots%
      \\ &%
           \leq%
           1 + \frac{1}{1!}  \pth{\frac{\alpha^2}{2}} + \frac{1}{2!}
           \pth{\frac{\alpha^2}{2}}^2 + \frac{1}{3!}
           \pth{\frac{\alpha^2}{2}}^3 + \cdots%
           =%
           \exp\pth{\alpha^2/2},
    \end{align*}
    as $(2i)! \geq 2^i {}i!$.

    We have that
    \begin{equation*}
        \tau%
        =%
        \Ex{e^{\alpha X_m}}%
        =%
        \Ex{ \Bigl.\smash{\prod_{i=1}^{m} e^{\alpha Y_i}}}
        =
        \Ex{ \Bigl.\smash{g(X_0, \ldots, X_{m-1})
              e^{\alpha Y_m}}},
        \qquad\text{where}\qquad
        g(X_0, \ldots, X_{m-1}) = \prod_{i=1}^{m-1} e^{\alpha Y_i}.
    \end{equation*}

    By the martingale property, we have that
    \begin{equation*}
        \ExCond{Y_i}{X_0,\ldots, X_{m-1}}
        =%
        \ExCond{Y_i}{g(X_0, \ldots, X_{m-1})}
        = 0.
    \end{equation*}
    By the above, this implies that
    \begin{math}
        \ExCond{ \smash{e^{\alpha Y_i}} }{ g(X_0, \ldots, X_{i-1}) }
        \leq \exp\pth{\alpha^2/2}.
    \end{math}
    Hence, by \lemref{expectations:product}, we have that
    \begin{align*}
      \tau
      =%
      \Ex{e^{\alpha X_m}}%
      &=%
        \Ex{ \Bigl.\smash{\prod_{i=1}^{m} e^{\alpha Y_i}}}%
        =%
        \Ex{ g(X_0,\ldots, X_{m-1})  e^{\alpha
        Y_m}} %
      \\[0.3cm]%
      &%
        =%
        \Ex{ \biggl. g(X_0,\ldots, X_{m-1})
        \ExCond{e^{\alpha Y_m}}{g(X_0, \ldots, X_{m-1})} }
        \leq%
        e^{\alpha^2/2} \Ex{ g(X_0, \ldots, X_{m-1})}
      \\[0.3cm]%
      &%
        \leq%
        \exp\pth[]{m \alpha^2/2}.
    \end{align*}

    Therefore, by Markov's inequality, we have
    \begin{align*}
        \Prob{ X_m > \lambda \sqrt{m}}%
        &=%
        \Prob{ e^{\alpha X_m} > e^{\alpha \lambda \sqrt{m}} } %
        =%
        \frac{\Ex{e^{\alpha X_m}}}{e^{\alpha \lambda \sqrt{m}}} =
        e^{m \alpha^2/2 -\alpha \lambda \sqrt{m}}
        \\ &%
        = \exp \pth{m
           (\lambda / \sqrt{m})^2/2 - (\lambda / \sqrt{m}) \lambda
           \sqrt{m}} = e^{-\lambda^2/2},
    \end{align*}
    implying the result.
\end{proof}

Here is an alternative form.

\begin{theorem}[Azuma's Inequality]
    Let $ X_0, \ldots, X_m$ be a martingale sequence such
    that and $|X_{i+1}-X_i| \leq 1$ for all $0 \leq i < m$.
    Let $\lambda > 0$ be arbitrary. Then
    \begin{math}
        \ds%
        \biggl.  \Prob{\cardin{X_m - X_0} > \lambda \sqrt{m}} <
        2\exp\pth{-\lambda^2/2}.
    \end{math}
\end{theorem}


\begin{example}
    Let $\chi(H)$ be the chromatic number of a graph
    $H$. What is chromatic number of a random graph? How
    does this random variable behaves?

    Consider the vertex exposure martingale, and let $X_i =
    \Ex{\chi(G) \sep{ G_i}}$. Again, without proving it, we claim
    that $X_0, \ldots, X_n=X$ is a martingale, and as such, we have:
    $\Prob{\cardin{X_n - X_0} > \lambda \sqrt{n}} \leq
    e^{-\lambda^2/2}$. However, $X_0 = \Ex{ \chi(G) }$, and $X_n =
    \Ex{ \chi(G) \sep{ G_n}} = \chi(G)$.  Thus,
    \begin{align*}
    \Prob{\cardin{\chi(G) - \Ex{\chi(G)}} > \lambda \sqrt{n}} \leq
    e^{-\lambda^2/2}.
    \end{align*}
    Namely, the chromatic number of a random graph is highly
    concentrated! And we do not even (need to) know what is the
    expectation of this variable!
\end{example}
