\section{The birthday paradox}

Consider a group of $n$ people, and assume their birthdays are
independently and uniformly distributed over the dates in the year (this
assumption is not quite true, but close enough). We are interested in the
question of how large $n$ has to be till we get a collision -- that is,
two people with the same birthday. Intuitively, since the year has
$m=365$ days\footnote{Leap years have $366$ days (every four years,
   roughly). Moon years have $354$ days, which means you have to add a
   leap month roughly every $19/7$ years, which is what the Jewish
   calendar does. If you do not do this, then the calendar is not
   synchronized with the weather (this is what the Islamic calendar
   does). Stick to the Gregorian calendar kids---it is much better.} , the
probability of a person to land on a specific birthday is $p=1/365$. So
the natural guess would be that $n$ needs to be approximately
$365$. Surprisingly, the answer is much smaller.

\begin{lemma}
    Let $X_1, \ldots, X_n$ be $n$ variables picked uniformly, randomly and
    independently from $\IRX{m}=\{1,\ldots, m\}$. Then, the expected number of
    collisions is $\binom{n}{2}/m$, where a \emphi{collision} is an
    unordered pair of indices with equal values.
\end{lemma}
\begin{proof}
    Let $Y_{i,j}=1$ $\iff$ $X_i = X_j$. We have that $\Ex{Y_{i,j}} =
    \Prob{Y_{i,j}=1} = 1/m$. Thus, the expected number of collisions
    is
    \begin{equation*}
        \Ex{\sum_{i=1}^{n-1} \sum_{j=i+1}^n  Y_{i,j}}
        =%
        \sum_{i=1}^{n-1} \sum_{j=i+1}^n  \Ex{Y_{i,j}}%
        =%
        \binom{n}{2} \frac{1}{m}.
    \end{equation*}
\end{proof}

\begin{remark}
    (A) For birthdays, for $m=365$, and $n=28$, we have that the expected
    number of collisions is
    \begin{equation*}
        \binom{28}{2} \frac{1}{365} = \frac{378}{365} > 1.
    \end{equation*}
    Namely, in expectation, in a room with 28 people we expect at least
    one pair of people to share a birthday.

    \smallskip (B) If a value is repeated $k > 1$ times, then there are
    $\binom{k}{2}$ associated collisions.
\end{remark}

An expected count above
one does not by itself establish a large probability of a collision.
The next lemma bounds that probability by bounding its complement.
\begin{lemma}
    Let $X_1, \ldots, X_n$ be $n$ variables picked uniformly, randomly
    and independently from $\IRX{m}=\{1,\ldots, m\}$. Then, the
    probability that no collision happened is at most
    $\exp\pth{ - \binom{n}{2}/m }$.
\end{lemma}
\begin{proof}
    If $n>m$, a collision is certain by the pigeonhole principle, and the
    bound is immediate. Hence assume $1\leq n\leq m$, and let $\EventB_0$ be
    the sure event (i.e., it is an empty event that always happens, and
    its probability is one). All conditioning events below then have
    positive probability.  Let $\Event_i$ be the event that $X_i$ is
    distinct from all the values in $X_1, \ldots, X_{i-1}$.  Let
    \begin{equation*}
        \EventB_k
        \EQ
        \Event_1 \cap \Event_2 \cap \cdots \cap \Event_k
%        \cap_{k=1}^i \Event_k
        \EQ \EventB_{k-1} \cap \Event_k
    \end{equation*}
    be the event that all of $X_1, \ldots, X_k$ are distinct.
    Clearly, we have
    \begin{align*}
        \ProbCond{\Event_i}{\EventB_{i-1}}
        &\EQ%
        \ProbCond{\Event_i}{\Event_1 \cap \cdots \cap \Event_{i-1}}
        \EQ%
        \frac{m-(i-1)}{m}
        \\&
        \EQ%
        1 - \frac{i-1}{m}
        \\&
        \LEQ
        \exp\pth{ - \frac{i-1}{m}}.
    \end{align*}

    Observe that
    \begin{align*}
        \Prob{\EventB_i}
        &\EQ
        \Prob{\Bigl.\,\smash{\overbrace{\Bigl.\Event_1 \cap \Event_2 \cap \cdots \cap \Event_{i-1}}^{\EventB_{i-1}} \cap \Event_i}}%
        \\&
        \EQ%
        \Prob{\EventB_{i-1}}
        \frac{\Prob{ \Event_i \cap \EventB_{i-1} }}{\Prob{\EventB_{i-1}}}
        \\[0.1cm]&
        \EQ%
        \Prob{\EventB_{i-1}}
        \ProbCond{\Event_i }{\EventB_{i-1}}
        \tag{By def of cond. prob}
        \\&
        \LEQ%
        \exp\pth{ - \frac{i-1}{m}}\Prob{\EventB_{i-1}}
        \\&
        \LEQ
        \prod_{k=1}^i
        \exp\pth{ - \frac{k-1}{m}}.
        \\&
        \EQ%
        \exp\pth{-\sum_{k=1}^i \frac{k-1}{m}}
        \\&
        \EQ%
        \exp\pth{ -\frac{i(i-1)}{2} \frac{1}{m}}
        \\&
        \EQ%
        \exp\pth{ -\binom{i}{2}/m}.
    \end{align*}
    This implies the desired claim for $i=n$.
\end{proof}

\section{Occupancy problems}

\begin{problem}
    We are throwing $m$ balls into $n$ bins randomly (i.e., for every
    ball we randomly and uniformly pick a bin independently of all other
    balls from the $n$ available bins, and place the ball in the bin
    picked).  There are many natural questions one can ask here:
    \begin{compactenumA}
        \smallskip%
        \item \itemlab{heavy_bin} What is the maximum number of balls in any bin?
        \smallskip%
        \item What is the number of bins which are empty?
        \smallskip%
        \item How many balls do we have to throw, such that all the
        bins are non-empty, with reasonable probability?
    \end{compactenumA}
\end{problem}

\begin{example}
    Consider throwing $n$ balls into $n$ bins, with $n\geq2$, and
    consider a specific ball. What is the probability that this ball is
    all alone in its own bin at the end of the process? Well, we might as
    well think about throwing this ball first, and all other balls later
    (by symmetry). Then, all it needs is that all other balls avoid its
    bin. The probability for that is
    \begin{equation*}
        p \EQ \pth{1- \frac{1}{n}}^{n-1} \;\approx\; \frac{1}{e}.
    \end{equation*}
    Thus, if $X_i$ is an indicator variable for the $i$\th ball being
    lonely, then the expected number of lonely balls (and thus the number
    of bins with exactly on ball) is
    \begin{equation*}
        \Ex{\sum_{i=1}^n X_i}
        \EQ%
        \sum_{i=1}^n \Ex{X_i}
        \EQ
        np
        \;\approx\;
        \frac{n}{e}.
    \end{equation*}
    Namely, a constant fraction of the balls are lonely in expectation.
\end{example}

\subsection{Understanding pileups}

The \emphi{load} of a bin is the number of balls in the bin. Here we are
interested in the maximum load---what is the maximum number of balls in a
single bin?

\subsubsection{Intuition}

\begin{observation}
    \obslab{empty_bin}%
    % 
    An alternative interesting question is how many bins are not empty
    (in expectation). The probability of a bin to be empty is
    $(1-1/n)^n \leq 1/e$, so by linearity of expectation the expected number
    of empty bins is $\leq n/e$.
\end{observation}

To understand the process better, imagine throwing in the balls. When we
throw a ball and it falls into a non-empty bin, we mark it as ``reject''
and put it into a set of rejected balls $R_1$. We do this until all balls
have been thrown. We now take all the rejected balls and throw them into
the bins, but restrict them to bins that contain at least one ball.
Again, if a ball falls into a bin that already has two balls, it is
rejected into a set $R_2$. In the $i$\th iteration of this process, all
the balls in $R_{i-1}$ are thrown into the bins that have exactly $i-1$
balls in them (no bin has more balls, since such balls were rejected into
$R_{i-1}$). Again, the process rejects all the balls that try to be put
into bins that already have exactly $i$ balls, putting them into the set
$R_i$. It is important at this point to convince yourself that this
process is indeed equivalent to the original process.

We are going to simplify things a bit---and thus get a loose bound, which
is intuitively correct (it is formally correct, but arguing its
correctness in detail is not worth the energy at this point).
Imagine that we are always throwing the $R_{i-1}$ balls into
$Y_{i-1}=|R_{i-1}|$ bins (the real number of bins being used would
typically be much bigger). By the above, we have
\begin{equation*}
\ExCond{Y_i}{Y_{i-1}}
\LEQ
Y_{i-1}/e.
\end{equation*}
To see the intuition behind this, observe that the number of bins that
still have only $i-1$ balls at the end of this process is, in expectation,
at most $Y_{i-1}/e$, by \obsref{empty_bin} (since all the bins under
consideration already have $i-1$ balls). Thus, in expectation, at least
\begin{equation*}
    \Delta_i \EQ (1-1/e)Y_{i-1}    
\end{equation*}
balls are used in creating the $i$\th layer, and hence
\begin{equation*}
    Y_i
    \LEQ
    Y_{i-1}-\Delta_i
    \EQ
    Y_{i-1}/e.    
\end{equation*}
Important! These calculations are really statements about expectations,
and we are being fast and loose with the math to keep the discussion
clear.  Thus,
\begin{align*}
    \Ex{Y_i}%
    &\EQ%
    \Ex{\Bigl.\ExCond{Y_i}{Y_{i-1}}}
    \LEQ
    \Ex{Y_{i-1}/e}
    \EQ
    \frac{1}{e}\Ex{Y_{i-1}}
    \&
    \LEQ
    \frac{1}{e^i}n.
\end{align*}
Namely, after $O(\log n)$ iterations, the expected number of rejected
balls becomes arbitrarily small. With high probability, after $O(\log n)$
iterations there are no rejected balls left. In other words, all bins
have load at most $O(\log n)$.


\subsubsection{Just doing the math}

One can prove a better bound, but that requires doing the exact calculations.

\begin{example}[Heavy times.]
    A ball is \emphi{$k$-heavy} if it there are $k-1$ or more balls that
    collide with it.  What is the expected number of $k$-heavy balls? For
    this to happen, there must be $k-1$ balls that ``conspire'' to
    collide with this ball.  The probability for that is $1/n^{k-1}$ for
    this specific group of $k-1$ balls, and if so, it creates one
    $k$-heavy ball. There are at most $\binom{n-1}{k-1}$ such groups, so
    by the union bound, the probability of a specific ball to be
    $k$-heavy is at most
    \begin{equation*}
        p_{k-1} = \frac{\binom{n-1}{k-1}}{n^{k-1}}.
    \end{equation*}
\end{example}

    $n$ bins, with $n$ sufficiently large.
    With probability at least $1- 1/n$, no bin has more than

    balls into $n$ bins (i.e., $m=n$). Clearly,
    \begin{equation*}
        \Ex{ X_i } = \sum_{j=1}^{n} \Prob{ \bigl. \text{The $j$\th

\subsection{Probability of all bins to %
   have exactly one ball}

Next, we are interested in the probability that all $m$ balls fall in
distinct bins.  Let $X_i$ be the event that the $i$\th ball fell in a
distinct bin from the first $i-1$ balls. We have:
\begin{align*}
  \Prob{\cap_{i=2}^{m} X_i }%
  &=%
    \Prob{X_2} \prod_{i=3}^{m} \ProbCond{X_i}{ \cap_{j=2}^{i-1}
    X_j}%
    \\&
    \leq%
    \prod_{i=2}^{m} \pth{\frac{n-i + 1}{n}}%
    \leq%
    \prod_{i=2}^{m}\pth{ 1 - \frac{i-1}{n}}%
  \\&%
  \leq%
  \prod_{i=2}^{m} e^{- \nfrac{(i-1)}{n}}%
  \leq%
  \exp \pth{ - \frac{m(m-1)}{2n} },
\end{align*}
thus for $m=\ceil{\smash{\sqrt{2n}} + 1}$, the probability that all
the $m$ balls fall in different bins is smaller than $1/e$.

This is sometimes referred to as the \emphw{birthday paradox}, which was
already mentioned above. You have $m=30$ people in the room, and you ask
them for the date (day and month) of their birthday (i.e., $n=365$). The
above shows that the probability of all birthdays to be distinct is at
most $\exp( -30\cdot 29/730 ) \leq 1/e$. Namely, there is more than
$50\%$ chance for a birthday collision, a simple but counter-intuitive
phenomenon.

\section{The Coupon Collector's Problem}

There are $n$ types of coupons, and at each trial one coupon is picked
independently and uniformly at random. How many trials does one have to
perform before picking all coupons?  Let $m$ be the number of trials
performed. We would like to bound the probability that $m$ exceeds a
certain number, and we still did not pick all coupons.

Let $C_i \in \brc{\bigl. 1, \ldots, n}$ be the coupon picked in the
$i$\th trial. The $j$\th trial is a success, if $C_j$ was not picked
before in the first $j-1$ trials. Let $X_i$ denote the number of trials
strictly after the $i$\th success, up to and including the $(i+1)$\th
success.  Let $X_0=1$.  Clearly, the number of trials performed is
\begin{align*}
    X = \sum_{i=0}^{n-1} X_i.
\end{align*}
Furthermore, $X_i$ has a geometric distribution with parameter $p_i$,
that is $X_i \sim \mathrm{Geom}(p_i)$, with $p_i = (n-i)/n$. The
expectation and variance of $X_i$ are
\begin{equation*}
    \Ex{X_i} = \frac{1}{p_i}
    \qquad\text{and}\qquad%
    \Var{X_i} = \frac{1-p_i}{p_i^2}.
\end{equation*}



\begin{lemma}
    Let $X$ be the number of rounds till we collect all $n$
    coupons. Then, $\Var{ X} \approx \pth{{\pi^2 }/{6}} n^2$ and its
    standard deviation is $\sigma_X \approx ( \pi / \sqrt{6} ) n$.
\end{lemma}
\begin{proof}
    The probability of $X_i$ to succeed in a trial is
    $p_i = \nfrac{(n-i)}{n}$, and $X_i$ has the geometric distribution
    with probability $p_i$. As such $\Ex{ X_i} = 1/p_i$, and
    $\Var{ X_i} = (1-p_i)/p_i^2$.  Thus,
    \begin{equation*}
        \Ex{ X}%
        =%
        \sum_{i=0}^{n-1} \Ex{  X_i}%
        =%
        \sum_{i=0}^{n-1} \frac{n}{n-i}%
        =%
        n H_n%
        =%
        n( \ln {n} + \Theta(1))%
        =%
        n \ln n + O(n),
    \end{equation*}
    where $H_n = \sum_{i=1}^{n} \nfrac{1}{i}$ is the $n$\th harmonic
    number.

 
    As for variance, using the independence of $X_0,\ldots, X_{n-1}$,
    we have
    \begin{align*}
      \Var{ X}%
      &\EQ%
        \sum_{i=0}^{n-1} \Var{ X_i}%        
        \\&
        \EQ%
        \sum_{i=0}^{n-1} \frac{1-p_i}{p_i^2}%
        \\&
        \EQ%
        \sum_{i=0}^{n-1} \frac{1-(n-i)/n}{\pth{\frac{n-i}{n}}^2}%
        \EQ%
        \sum_{i=0}^{n-1} \frac{i/n}{\pth[]{\frac{n-i}{n}}^2}%
        \\&
        \EQ
        \sum_{i=0}^{n-1} \frac{i}{n} \pth{\frac{n}{n-i}}^2%
      \\&%
      \EQ%
      n \sum_{i=0}^{n-1} \frac{i}{(n-i)^2}%
      =%
      n \sum_{i=1}^{n} \frac{n-i}{i^2}%
      \\&
      \EQ% 
      n \pth[]{ \sum_{i=1}^{n} \frac{n}{i^2} - \sum_{i=1}^{n}
      \frac{1}{i}}%
      =%
      n^2 \sum_{i=1}^{n} \frac{1}{i^2} - n H_n
      \approx %
      \frac{\pi^2}{6}n^2,
    \end{align*}
    since
    $\lim_{n\rightarrow \infty} \sum_{i=1}^{n} \frac{1}{i^2} =
    \pi^2/6$, we have
    $\displaystyle \lim_{n\rightarrow \infty} \frac{\Var{X}}{n^2} =
    \frac{\pi^2}{6}$.
\end{proof}

This implies a weak bound on the concentration of $X$, using Chebyshev's
inequality. We have
\begin{align*}
  \Prob{ X \geq n \ln n + n + t \cdot n \smash{\frac{\pi}{\sqrt{6}}}\Bigr. }%
  \leq%
  \Prob{ \Bigl. \cardin{ \bigl. X - \Ex{X}} \geq t \sigma_X } \leq
  \frac{1}{t^2},
\end{align*}
