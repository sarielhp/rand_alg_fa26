\begin {defn}[\DefName {Convex hull}] The \emphi {convex hull} of a set $\PntSetC \subseteq \Re ^\hdim $ is the set of all convex combinations of points of $\PntSetC $; that is, \[ \CHX { \PntSetC } = \brc {\sum _{i=0}^m \alpha _i \pntB _i\sep { \forall i \;\;\pntB _i \in \PntSetC , \alpha _i \geq 0, \text { and } \sum _{j=1}^m \alpha _i =1}}. \] \deflab {convex:hull} \end {defn}
