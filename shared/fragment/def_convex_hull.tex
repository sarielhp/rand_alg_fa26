\begin{defn}[\DefName{Convex hull}]
        The \emphOnly{convex hull}\index{convex hull} of a set $\PntSetC \subseteq
        \Re^\hdim$ is the set of all convex combinations of points of
        $\PntSetC$; that is,
        \[
        \CHX{ \PntSetC} = \brc{\sum_{i=0}^m \alpha_i \pntB_i\sep{
              \forall i \;\;\pntB_i \in \PntSetC, \alpha_i \geq 0,
              \text{ and } \sum_{j=1}^m \alpha_i =1}}.
        \]
        \deflab{convex:hull}
    \end{defn}
