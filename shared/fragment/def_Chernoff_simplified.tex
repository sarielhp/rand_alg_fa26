\begin{theorem}
        \thmlab{Chernoff:simplified}%
        \index{Chernoff inequality}%
        \index{Chernoff inequality!simplified form}%
        %
        Let $X_1, \ldots, X_n$ be $n$ independent variables, where
        \begin{equation*}
            \Prob{ \bigl. X_i = 1 } =p_i
            \qquad\text{and}\qquad
            \Prob{ \bigl. X_i = 0 } =q_i = 1-p,
        \end{equation*}
        for all $i$.  Let $X = \sum_{i=1}^{n} X_i$ and
        $\mu = \Ex{\bigl. X} = \sum_i p_i$.  For any $\delta > 0$, we have
        \begin{equation*}
            \Prob{ \bigl. X > (1+\delta)\mu } < \pth{
               \NiceFrac{e^\delta}{(1+\delta)^{1+\delta}}}^\mu.
        \end{equation*}
