\begin{lemma}
    \lemlab{unique:inverse}%
    %
    Consider a prime $p$, and any numbers $x, y \in \ZZ_p$.  If
    $x \neq y$ then, for each pair of numbers
    $r,s \in \ZZ_{p} = \{0,1, \ldots, p-1\}$,
    there is exactly one unique choice of numbers $a,b \in \ZZ_p$ such
    that $ax + b \equiv r \pmod p$ and $ay + b \equiv s \pmod p$.
\end{lemma}
