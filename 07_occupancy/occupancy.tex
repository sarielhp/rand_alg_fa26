\section{Some needed math}


\begin{fragment}{d:approx:basic}
\begin{lemma}
    \lemlab{approximation:basic}%
    %
    For any positive integer $n$, we have:
    \begin{compactenumi}
        \medskip%
        \item $1+x \leq e^x$ \;\;and\;\; $1-x \leq e^{-x}$, for all $x$.

        \medskip
        \item $(1+1/n)^n \LEQ e \LEQ \bigl(1+1/n\bigr)^{n+1}$.

        \medskip
        \item $(1-1/n)^n \leq \frac{1}{e} \leq (1-1/n)^{n-1} $.

        \medskip
        \item $(n/e)^n \LEQ n! \LEQ (n+1)^{n+1} / e^n$.

        \item For any $k \leq n$, we have:
        \begin{math}
            \ds%
            \pth[]{\frac{n}{k}}^k
            \LEQ%
            \binom{n}{k}%
            \LEQ%
            \pth[]{\frac{n e}{k}}^k.
        \end{math}
    \end{compactenumi}
\end{lemma}
\end{fragment}

\begin{proof}
    \textbf{(i)} Let $h(x) = e^x - 1 - x$. Observe that
    $h'(x) = e^x - 1$, and $h''(x) = e^x > 0$, for all $x$. That is
    $h(x)$ is a convex function. It achieves its minimum at
    $h'(x) = 0$ $\implies$ $e^x = 1$, which is true for $x=0$. For
    $x=0$, we have that $h(0) =e^0- 1-0 = 0$. That is, $h(x) \geq 0$
    for all $x$, which implies that $e^x \geq 1+x$, see
    \figref{exp:1:plus:x}.


    \begin{figure}[t]
        \centerline{\includegraphics{\File{figs/exp_linear}}}%
        \caption{}
        \figlab{exp:1:plus:x}
    \end{figure}

    \medskip%
    \noindent%
    \textbf{(ii, iii)} Indeed, $1+1/n\leq \exp \pth{1/n}$ and
    $(1-1/n)^n \leq \exp(-1/n)$, by (i). As such
    \begin{equation*}
        (1+1/n)^n \leq \exp\pth{n(1/n)} = e
        \qquad\text{ and }\qquad%
        (1-1/n)^n \leq \exp\pth{n(-1/n)} = \frac{1}{e},
    \end{equation*}
    which implies the left sides of (ii) and (iii).  These are
    equivalent to
    \begin{equation*}
        \frac{1}{e} \leq \pth{\frac{n}{n+1}}^n%
        =%
        \pth{ 1 - \frac{1}{n+1}}^n
        \qquad\text{ and }\qquad%
        e  \leq \pth{1 + \frac{1}{n-1}}^n,
    \end{equation*}
    which are the right side of (iii) [by replacing $n+1$ by $n$], and
    the right side of (ii) [by replacing $n$ by $n+1$].


    \medskip%
    \noindent%
    \textbf{(iv)} Indeed,
    \begin{align*}
        \frac{n^n}{n!} \leq \sum_{i=0}^{\infty} \frac{n^i}{i!} = e^n,
    \end{align*}
    by the Taylor expansion of $e^x = \sum_{i=0}^\infty
    \frac{x^i}{i!}$. This implies that $(n/e)^n \leq n!$, as required.

    As for the righthand side. The claim holds for $n=0$ and
    $n=1$. Let $f(n) = (n+1)^{n+1}/e^n$, and observe that by (ii), we
    have
    \begin{equation*}
        \frac{f(n)}{f(n-1)}
        =%
        \frac{(n+1)^{n+1} / e^n}{n^n / e^{n-1}}
        =%
        \frac{n}{e}\pth{\frac{n+1}{n}}^{n+1}
        =%
        \frac{n}{e}\pth{1 + \frac{1}{n}}^{n+1}
        \geq
        n \frac{e}{e} = n.
    \end{equation*}
    Thus, we have
    \begin{equation*}
        \frac{(n+1)^{n+1}}{e^n}
        =
        f(n)
        =
        \frac{f(n)}{f(n-1)} \cdot
        \frac{f(n-1)}{f(n-2)} \cdots
        \frac{f(1)}{f(0)}
        \geq
        n \cdot n-1 \cdots 1 = n!
    \end{equation*}


    \medskip%
    \noindent%
    \textbf{(v)} For any $k \leq n$, we have
    $\frac{n}{k} \leq \frac{n-1}{k-1}$ since
    $kn -n = n(k-1) \leq k(n-1) = kn - k$. As such,
    $\frac{n}{k} \leq \frac{n-i}{k-i}$, for $1 \leq i \leq k-1$. As
    such,
    \begin{align*}
        \pth{\frac{n}{k}}^k \leq \frac{n}{k} \cdot \frac{n-1}{k-1}
        \cdots \frac{n-i}{k-i} \cdots \frac{n-k+1} {1} =
        \frac{n!}{(n-k)! k!} = \binom{n}{k}.
    \end{align*}
    As for the other direction, we have
    \begin{math}
        \ds
      \binom{n}{k} \leq \frac{n^k}{k!} \leq
      \frac{n^k}{\pth{{k}/{e}}^{k}} = \pth{ \frac{n e}{k} }^k,
    \end{math}
    %\begin{align*}
    %\end{align*}
    by (iii).
\end{proof}

\section{The birthday paradox}

Consider a group of $n$ people, and assume their birthdays are
uniformly distributed no the dates in the year (this assumption is not
quite true, but close enough). We are interested in the question of
how large $n$ has to be  till we get a collision -- that is, two
people with the same birthday. Intuitively, since the year has $m=364$
days, the probability of person to land on a specific birthday is
$p=1/364$. So the natural guess would be that $n$ needs to be
approximately $364$. Surprisingly, the answer is much smaller.

\begin{lemma}
    Let $X_1, \ldots, X_n$ be $n$ variables picked uniformly, randomly
    and independently from $\IRX{m}=\{1,\ldots, m\}$. Then, the
    expected number of collisions is $\binom{n}{2}/m$.
\end{lemma}
\begin{proof}
    Let $Y_{i,j}=1$ $\iff$ $X_i = X_j$. We have that $\Ex{Y_{i,j}} =
    \Prob{Y_{i,j}=1} = 1/m$. Thus, the expected number of collisions
    is
    \begin{equation*}
        \Ex{\sum_{i=1}^{n-1} \sum_{j=i+1}^n  Y_{i,j}}
        =%
        \sum_{i=1}^{n-1} \sum_{j=i+1}^n  \Ex{Y_{i,j}}%
        =%
        \binom{n}{2} \frac{1}{m}.
    \end{equation*}
\end{proof}
As such, for birthdays, for $m=364$, and $n=28$, we have that the
expected number of collisions is
\begin{equation*}
    \binom{28}{2} \frac{1}{364} = \frac{378}{364} > 1.
\end{equation*}

This seems weird, but is it the truth?
\begin{lemma}
    Let $X_1, \ldots, X_n$ be $n$ variables picked uniformly, randomly
    and independently from $\IRX{m}=\{1,\ldots, m\}$. Then, the
    probability that no collision happened is at most
    $\exp\pth{ - \binom{n}{2}/m }$.
\end{lemma}
\begin{proof}
    Let $\Event_i$ be the event that $X_i$ is distinct from all the
    values in $X_1, \ldots, X_{i-1}$.  Let
    \begin{equation*}
        \EventB_k
        \EQ
        \Event_1 \cap \Event_2 \cap \cdots \cap \Event_k
%        \cap_{k=1}^i \Event_k
        \EQ \EventB_{k-1} \cap \Event_k
    \end{equation*}
    be the event that all of $X_1, \ldots, X_i$ are distinct.
    Clearly, we have
    \begin{align*}
        \ProbCond{\Event_i}{\EventB_{i-1}}
        &\EQ%
        \ProbCond{\Event_i}{\Event_1 \cap \cdots \cap \Event_{i-1}}
        \EQ%
        \frac{m-(i-1)}{m}
        \\&
        \EQ%
        1 - \frac{i-1}{m}
        \\&
        \LEQ
        \exp\pth{ - \frac{i-1}{m}}.
    \end{align*}

    Observe that
    \begin{align*}
        \Prob{\EventB_i}
        &\EQ
        \Prob{\Bigl.\,\smash{\overbrace{\Bigl.\Event_1 \cap \Event_2 \cap \cdots \cap \Event_{i-1}}^{\EventB_{i-1}} \cap \Event_i}}%
        \\&
        \EQ%
        \Prob{\EventB_{i-1}}
        \frac{\Prob{ \Event_i \cap \EventB_{i-1} }}{\Prob{\EventB_{i-1}}}
        \\[0.1cm]&
        \EQ%
        \Prob{\EventB_{i-1}}
        \ProbCond{\Event_i }{\EventB_{i-1}}
        \tag{By def of cond. prob}
        \\&
        \LEQ%
        \exp\pth{ - \frac{i-1}{m}}\Prob{\EventB_{i-1}}
        \\&
        \LEQ
        \prod_{k=1}^i
        \exp\pth{ - \frac{k-1}{m}}.
        \\&
        \EQ%
        \exp\pth{-\sum_{k=1}^i \frac{k-1}{m}}
        \\&
        \EQ%
        \exp\pth{ -\frac{i(i-1)}{2} \frac{1}{m}}
        \\&
        \EQ%
        \exp\pth{ -\binom{i}{2}/m}.
    \end{align*}
    Which implies the desired claim for $i=n$.
\end{proof}

\section{Occupancy problems}

\begin{problem}
    We are throwing $m$ balls into $n$ bins randomly (i.e., for every
    ball we randomly and uniformly pick a bin from the $n$ available
    bins, and place the ball in the bin picked).
    There are many natural questions one can ask here:
    \begin{compactenumA}
        \item What is the maximum number of balls in any bin?
        \item What is the number of bins which are empty?
        \item How many balls do we have to throw, such that all the
        bins are non-empty, with reasonable probability?
    \end{compactenumA}
\end{problem}

\begin{example}
    Consider throwing $n$ into $n$ bins, and consider a specific
    ball. What is the probability that this ball all alone in its own bin
    in the end of the process? Well, we might as well think about throwing this ball first, and all ball later (by symmetry). Then, all it needs is that all other balls avoid its bin. The probability for that is
    \begin{equation*}
        p \EQ \pth{1- \frac{1}{n}}^{n-1} \;\approx\; \frac{1}{e}.
    \end{equation*}
    Thus, if $X_i$ is an indicator variable for the $i$\th ball being
    lonely, then the expected number of empty balls is
    \begin{equation*}
        \Ex{\sum_{i=1}^n X_i}
        \EQ%
        \sum_{i=1}^n \Ex{X_i}
        \EQ
        np
        \;\approx\;
        \frac{n}{e}.
    \end{equation*}
    Namely, a constant fraction of the balls are lonely.
\end{example}

\begin{theorem}
    With probability at least $1- 1/n$, no bin has more than
    $\displaystyle k^* = \ceil{\frac{3\ln{n}}{\ln \ln n}}\;$ balls in
    it.
\end{theorem}
\begin{proof}
    Let $X_i$ be the number of balls in the $i$\th bins, when we throw $n$
    balls into $n$ bins (i.e., $m=n$). Clearly,
    \begin{equation*}
        \Ex{ X_i } = \sum_{j=1}^{n} \Prob{ \bigl. \text{The $j$\th
              ball fall in $i$\th bin}} = n \cdot \frac{1}{n} = 1,
    \end{equation*}
    by linearity of expectation.  The probability that the first bin has
    exactly $i$ balls is
    \begin{align*}
        p_i
        &\EQ
        \binom{n}{i}\, \pth{\frac{1}{n}}^i \pth[]{ 1 -
      \frac{1}{n}}^{n-i}%
      \LEQ%
      \binom{n}{i}\, \pth[]{\frac{1}{n}}^i%
      \\&%
      \LEQ%
      \pth[]{\frac{n e}{i}}^i \pth[]{\frac{1}{n}}^i%
      \EQ%
      \pth[]{\frac{e}{i}}^{i}
    \end{align*}
    This follows by \lemref{approximation:basic} (iv).

    Let $C(k)$ be the event that the first bin (but really any specific
    bin, as they all behave in the same way by symmetry) has $k$ or more
    balls in it. Then,
    \begin{align*}
      \Prob{\bigl. C(k)}%
      &\LEQ%
      \sum_{i=k}^{n} p_i%
      \LEQ
      \sum_{i=k}^{n} \pth[]{\frac{e}{i}}^{i}%
      \LEQ%
      \pth[]{\frac{e}{k}}^{k} \pth{1 + \frac{e}{k} + \frac{e^2}{k^2} +
      \ldots }%
   \\&
   \EQ%
      \pth[]{\frac{e}{k}}^k \frac{1}{1-e/k}.
    \end{align*}

    For $k^* = c \ln{n} /\ln \ln n$, we have
    \begin{align*}
      \Prob{\bigl. C\pth{k^*}}%
      &\leq%
        \pth{\bigl.\frac{e}{k^*}}^{k^*} \frac{1}{1-e/{k^*}}%
        \leq%
        2 \exp \pth{\Bigl. k^* \pth{1  - \ln k^* }}
        \leq%
        2 \exp \pth{ - \frac{k^*\ln k^*}{2} }
      \\&%
      \leq%
      2 \exp \pth{ - \frac{c \ln{n}}{2 \ln \ln n}
      \smash{\underbrace{\ln
      \frac{c \ln{n}}{\ln \ln n}}_{\approx \ln \ln n}}}
      \leq %
      2 \exp \pth{ - \frac{c \ln{n}}{4 } }
      \leq %
      \frac{1}{n^{c/4}},
      \leq %
      \frac{1}{n^{2}},
    \end{align*}
    % \bigskip%
    for $n$ and $c\geq 8$. % \bigskip%

    Let us redo this calculation more carefully (yuk!). For
    \begin{math}
        k^* = \ceil{
           \tfrac{3 \ln n }{\ln \ln n}
           },
    \end{math}
    we have
    \begin{align*}
        \Prob{\bigl. C\pth{k^*}}%
        &\LEQ%
        \pth{\frac{e}{k^*}}^{k^*} \frac{1}{1-e/{k^*}}%
        \\&
        \LEQ%
        2 \pth[]{\frac{e}{(3 \ln{n})/\ln \ln n}}^{k^*}%
        \\&
        \EQ%
        2  \,{ \bigl. \exp\pth{\Bigl.
              \smash{\underbrace{1- \ln 3}_{<0}} - \ln \ln {n} + \ln \ln
              \ln {n}}}^{k^*}%
        \\&%
        \LEQ%
        2 \bigl.\exp \pth{\bigl. \pth{ - \ln \ln {n} + \ln \ln \ln
              {n}}{k^*}}
        \\&
        \LEQ%
        2 \exp \pth{-3 \ln n + {6 \ln n} \frac{\ln \ln \ln n}{\ln \ln n}}
        \\&%
        \LEQ%
        2\exp \pth{ -2.5 \ln n}
        \\&
        \LEQ%
        \frac{1}{n^2},
    \end{align*}
    for $n$ large enough. We conclude, by the union bound, that since
    there are $n$ bins and they have identical distributions that
    \begin{equation*}
        \Prob{\MakeBig \text{any bin contains more than $k^*$
              balls}} %
        \leq%
        \sum_{i=1}^{n} C_{i}\pth{k^*}%
        \leq%
        \frac{1}{n}.
    \end{equation*}
\end{proof}


\begin{exercise}
    Show that when throwing $m=n \ln n$ balls into $n$ bins, with
    probability $1 - o(1)$, every bin has $O(\log{n})$ balls.
\end{exercise}



\subsection{Probability of all bins to %
   have exactly one ball}

Next, we are interested in the probability that all $m$ balls fall in
distinct bins. Let $X_i$ be the event that the $i$\th ball fell in a
distinct bin from the first $i-1$ balls. We have:
\begin{align*}
  \Prob{\cap_{i=2}^{m} X_i }%
  &=%
    \Prob{X_2} \prod_{i=3}^{m} \ProbCond{X_i}{ \cap_{j=2}^{i-1}
    X_j}%
    \leq%
    \prod_{i=2}^{m} \pth{\frac{n-i + 1}{n}}%
    \leq%
    \prod_{i=2}^{m}\pth{ 1 - \frac{i-1}{n}}%
  \\&%
  \leq%
  \prod_{i=2}^{m} e^{- \nfrac{(i-1)}{n}}%
  \leq%
  \exp \pth{ - \frac{m(m-1)}{2n} },
\end{align*}
thus for $m=\ceil{\smash{\sqrt{2n}} + 1}$, the probability that all
the $m$ balls fall in different bins is smaller than $1/e$.

This is sometime referred to as the \emphw{birthday paradox}, which
was already mentioned above. You have $m=30$ people in the room, and
you ask them for the date (day and month) of their birthday (i.e.,
$n=365$). The above shows that the probability of all birthdays to be
distinct is $\exp( -30\cdot 29/730 ) \leq 1/e$. Namely, there is more
than $50\%$ chance for a birthday collision, a simple but
counter-intuitive phenomena.

\section{The Coupon Collector's Problem}

There are $n$ types of coupons, and at each trial one coupon is picked
in random. How many trials one has to perform before picking all
coupons?  Let $m$ be the number of trials performed. We would like to
bound the probability that $m$ exceeds a certain number, and we still
did not pick all coupons.

Let $C_i \in \brc{\bigl. 1, \ldots, n}$ be the coupon picked in the
$i$\th trial. The $j$\th trial is a success, if $C_j$ was not picked
before in the first $j-1$ trials. Let $X_i$ denote the number of
trials from the $i$\th success, till after the $(i+1)$\th
success. Clearly, the number of trials performed is
\begin{align*}
    X = \sum_{i=0}^{n-1} X_i.
\end{align*}
Furthermore, $X_i$ has a geometric distribution with parameter $p_i$,
that is $X_i \sim \mathrm{Geom}(p_i)$, with $p_i = (n-i)/n$. The
expectation and variance of $X_i$ are
\begin{equation*}
    \Ex{X_i} = \frac{1}{p_i}
    \qquad\text{and}\qquad%
    \Var{X_i} = \frac{1-p_i}{p_i^2}.
\end{equation*}



\begin{lemma}
    Let $X$ be the number of rounds till we collection all $n$
    coupons. Then, $\Var{ X} \approx \pth{{\pi^2 }/{6}} n^2$ and its
    standard deviation is $\sigma_X \approx ( \pi / \sqrt{6} ) n$.
\end{lemma}
\begin{proof}
    The probability of $X_i$ to succeed in a trial is
    $p_i = \nfrac{(n-i)}{n}$, and $X_i$ has the geometric distribution
    with probability $p_i$. As such $\Ex{ X_i} = 1/p_i$, and
    $\Var{ X_i} = q/p^2 = (1-p_i)/p_i^2$.

    Thus,
    \begin{equation*}
        \Ex{ X}%
        =%
        \sum_{i=0}^{n-1} \Ex{  X_i}%
        =%
        \sum_{i=0}^{n-1} \frac{n}{n-i}%
        =%
        n H_n%
        =%
        n( \ln {n} + \Theta(1))%
        =%
        n \ln n + O(n),
    \end{equation*}
    where $H_n = \sum_{i=1}^{n} \nfrac{1}{i}$ is the $n$\th Harmonic
    number.

    As for variance, using the independence of $X_0,\ldots, X_{n-1}$,
    we have
    \begin{align*}
      \Var{ X}%
      &\EQ%
        \sum_{i=0}^{n-1} \Var{ X_i}%        
        \\&
        \EQ%
        \sum_{i=0}^{n-1} \frac{1-p_i}{p_i^2}%
        \\&
        \EQ%
        \sum_{i=0}^{n-1} \frac{1-(n-i)/n}{\pth{\frac{n-i}{n}}^2}%
        \EQ%
        \sum_{i=0}^{n-1} \frac{i/n}{\pth[]{\frac{n-i}{n}}^2}%
        \\&
        \EQ
        \sum_{i=0}^{n-1} \frac{i}{n} \pth{\frac{n}{n-i}}^2%
      \\&%
      \EQ%
      n \sum_{i=0}^{n-1} \frac{i}{(n-i)^2}%
      =%
      n \sum_{i=1}^{n} \frac{n-i}{i^2}%
      \\&
      \EQ% 
      n \pth[]{ \sum_{i=1}^{n} \frac{n}{i^2} - \sum_{i=1}^{n}
      \frac{1}{i}}%
      =%
      n^2 \sum_{i=1}^{n} \frac{1}{i^2} - n H_n
      \approx %
      \frac{\pi^2}{6}n^2,
    \end{align*}
    since
    $\lim_{n\rightarrow \infty} \sum_{i=1}^{n} \frac{1}{i^2} =
    \pi^2/6$, we have
    $\displaystyle \lim_{n\rightarrow \infty} \frac{\Var{X}}{n^2} =
    \frac{\pi^2}{6}$.
\end{proof}

This implies a weak bound on the concentration of $X$, using Chebyshev
inequality, we have
\begin{align*}
  \Prob{ X \geq n \ln n + n + t \cdot n \smash{\frac{\pi}{\sqrt{6}}}\Bigr. }%
  \leq%
  \Prob{ \Bigl. \cardin{ \bigl. X - \Ex{X}} \geq t \sigma_X } \leq
  \frac{1}{t^2},
\end{align*}

Note, that this is somewhat approximate, and hold for $n$ sufficiently
large.
