\section{Some useful estimates}

\begin{lemma}
    For any $n \geq 2$, and $m \geq 1$, we have that $(1-1/n)^m\geq 1
    - m/n$.
\end{lemma}
\begin{proof}
    Follows by induction. Indeed, for $m=1$ the claim is
    immediate. For $m \geq 2$, we have
    \begin{align*}
        \pth{1-\frac{1}{n}}^m%
        =%
        \pth[]{1-\frac{1}{n}} \pth[]{1-\frac{1}{n}}^{m-1}%
        \geq%
        \pth[]{1-\frac{1}{n}}\pth[]{ 1 - \frac{m-1}{n}} \geq%
        1 - \frac{m}{n}.
    \end{align*}
\end{proof}

This implies the following.

\begin{lemma}
    For any $m\leq n$, we have that $1-m/n \leq (1-1/n)^m \leq
    \exp\pth{-m/n}$.
\end{lemma}
