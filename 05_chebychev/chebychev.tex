\section{Chebyshev's inequality}

Here, we discuss Chebyshev's inequality, sometimes written as Chebychev's
inequality using the French spelling of Chebyshev\footnote{\emph{Pafnuty
      Lvovich Chebyshev} (Russian: \SpellIgnore{Пафну́тий Льво́вич
      Чебышёв}; May 16, 1821--December 8, 1894) was a Russian
   mathematician and is considered to be the founding father of Russian
   mathematics; see~\cite{b-pct-99}. The Russian letter \emph{ё} in
   \SpellIgnore{Чебышёв} is pronounced ``yo'', making the native
   pronunciation ``\SpellIgnore{Cheby-SHOV}'', rhyming with ``shove''.}
% \remX{Historical priority: The inequality was first discovered and
%    published in 1853 by the French mathematician Ir{\'e}n{\'e}e-Jules
%    Bienaym{\'e}~\cite{b-cadld-1853}, fourteen years before Chebyshev
%    published it independently in 1867~\cite{c-dvm-1867}. Andrei Markov
%    later acknowledged Bienaym{\'e}'s priority, which is why it is often
%    called the Bienaym{\'e}--Chebyshev inequality; see Butzer and
%    Jongmans~\cite{b-pct-99}.}


\subsection{Example: A better inequality via moments}


Let $X_1, \ldots, X_n \in \brc{-1,+1}$ be mutually independent random variables,
each taking each value in $\brc{-1,+1}$ with equal probability $1/2$. Let
$Y= \sum_{i=1}^n X_i$. We have that
\begin{equation*}
    \Ex{Y\bigr.}%
    \EQ%
    \Ex{\Bigl. \smash{\sum_i} X_i}%
    \EQ%
    \smash{\sum_i} \Ex{X_i \bigr.}%
    \EQ%
    n \cdot 0%
    \EQ
    0.
\end{equation*}

A more interesting quantity is
\begin{align*}
  \Ex{Y^2} %
  &\EQ%
    \Ex{ \smash{\Bigl(\smash{\sum_i} X_i\Bigr)^2} \Bigr.}%
 %   \\&
    \EQ%
    \Ex{\Bigl. \smash{\sum_i} X_i^2 + 2 \smash{\sum_{i < j}} X_i X_j}%
    \\[.2cm]&
    \EQ%
    \sum_i \Ex{  X_i^2} + 2\Ex{  \smash{\sum_{i < j}} X_i X_j \Bigr.}%
    \\&
    \EQ%
    n  + 2 \sum_{i < j} \Ex{ X_i X_j}%
  \\&%
  \EQ%
  n + 2 \sum_{i < j} \Ex{ X_i} \Ex{ X_j}%
  \EQ%
  n.
\end{align*}

\begin{remark}[Maybe skip for now]
    Note that the last step in the above derivation requires only
    pairwise independence of $X_1, \ldots, X_n$, rather than full mutual
    independence. This property is a principal advantage of Chebyshev's
    inequality over higher-moment techniques (e.g., Chernoff's
    inequality). This comment might make sense only later in the
    course---put differently, Chebyshev is NOT strictly dominated by
    other tools.
\end{remark}

\begin{lemma}
    Let $X_1, \ldots, X_n \in \brc{-1,+1}$ be independent random variables, each
    taking each value in $\brc{-1, +1}$ with probability $1/2$. For any
    $t > 0$, we have that
    \begin{equation*}
        \Prob{ \cardin{\sum_i X_i} \geq t \sqrt{n}} \leq 1/t^2.
    \end{equation*}
\end{lemma}

\begin{proof}
    Let $Y= \sum_i X_i$ and $Z =Y^2$. Notice that $Z \geq 0$ with
    $\Ex{Z} = n$.  We have
    \begin{align*}
        \Prob{ \cardin{\Bigl.\smash{\sum_i} X_i} \geq t \sqrt{n}}%
        &=%
        \Prob{ \smash{ \Bigl(\sum_i X_i\Bigr)^2 \geq t^2 n} \Bigr.}%
        =%
        \Prob{ Y^2 \geq t^2 \Ex{Y^2} \Bigr.} %
        \\&=%
        \Prob{ Z \geq t^2 \Ex{Z}} %
        \leq%
        \frac{1}{t^2},
    \end{align*}
    by Markov's inequality.
\end{proof}


\subsection{Chebyshev's inequality}

As a reminder, the variance of a random variable $X$ is
\begin{equation*}
    \Var{ X} = \Ex{ (X -\mu_X)^2 } = \Ex{X^2} - \mu_X^2,
\end{equation*}
where $\mu_X = \Ex{X}$.




\begin{fragment}{\deflabel{Chebyshev}}
    \begin{theorem}[Chebyshev's inequality]
        \thmlab{Chebyshev:inequality}%
        %
        Let $X$ be a real random variable, with $\mu_X = \Ex{X}$, and
        $\sigma_X = \sqrt{\Var{X}}$. Then, for any $t > 0$, we have
        \begin{equation*}
            \Prob{\bigl. \cardin{X - \mu_X} \geq t \sigma_X} \leq
            \frac{1}{t^2}.
        \end{equation*}
        % where $\mu_X = \Ex{\bigl. X}$ and
        % $\sigma_X = \sqrt{\Var{ X}}$.
    \end{theorem}
\end{fragment}



\begin{proof}
    Set $Y = \pth{X - \mu_X}^2$, and observe that
    \begin{equation*}
        \sigma_X^2
        =%
        \Var{X} =
        \Ex{ Y}
        =%
        \Ex{\pth{X - \mu_X}^2}
        =%
        \Ex{X^2}  - \mu_X^2.
    \end{equation*}
    As such, we have that
    \begin{equation*}
        \Prob{\bigl. \cardin{X - \mu_X}\geq t \sigma_X}%
        =%
        \Prob{\,\pth{X - \mu_X}^2\geq t^2 \sigma_X^2}
        =%
        \Prob{ Y \geq t^2 \Ex{Y}\bigl.\!}
        \leq%
        \frac{1}{t^2},
    \end{equation*}
    by \thmrefY{Markov's inequality}{Markov}.
\end{proof}

\begin{exercise}[Not too interesting]
    Consider the case that $\sigma_X = \Var{X} = 0$, what happens then
    with Chebyshev's inequality?
\end{exercise}

\begin{example}[Variance of a binomial distribution]
    Let $Y = \sum_{i=1}^m Y_i$, where $Y_1, \ldots, Y_m$ are independent random
    $0/1$ variables that are each one with probability $p$ (i.e., a
    single $Y_i$ is a Bernoulli random variable taking value $1$ with
    probability $p$ and $0$ with probability $1-p$). The random variable
    $Y$ has a binomial distribution with parameters $m$ and $p$.  That
    is, $Y \sim \DistBinom{m}{p}$. Thus, for all $i$, we have
    $\Ex{Y_i} = p$, and
    $\Var{Y_i} = \Ex{Y_i^2} - \pth{\Ex{Y_i}}^2 = p - p^2 = p(1-p)$. Since
    variance is additive for pairwise independent random variables, we
    have
    \begin{equation}
        \Var{Y} = \Var{\Bigl. \smash{\sum_i Y_i}} =
        \smash{\sum_{i=1}^m \Var{Y_i}} = m p (1-p).
        \eqlab{var-bionomial}
    \end{equation}
    This also implies that
    $\Ex{\smash{Y^2}} = \Var{Y} + \pth{\Ex{Y}}^2 = m p (1-p) + m^2 p^2$.
\end{example}

\Section{Estimation via sampling}

One of the main advantages of randomized algorithms is that they sample
the world---that is, learn what the input looks like without reading all
the input. For example, consider the following problem: We are given a
set $U$ of $n$ objects $u_1, \ldots, u_n$, and we want to compute the number
of elements of $U$ that have some property. Assume that one can check if
this property holds, in constant time, for a single object, and let
$\propertyX{u}$ be the function that returns $1$ if the property holds
for the element $u$, and $0$ otherwise.  Now, let $\numC$ be the number
of objects in $U$ that have this property. We want to reliably estimate
$\numC$ without computing the property for all the elements of $U$.

A natural approach is to pick a random sample $\Sample$ of $m$ objects
$r_1, \ldots, r_m \in U$ (with replacement), and compute
$Y = \sum_{i=1}^m \propertyX{r_i}$.  The estimate for $\numC$ is
$Z = (n/m) Y$. It is natural to ask how far $Z$ is from the true value
$\numC$.


\begin{lemma}
    \lemlab{Chebyshev-1}
    %
    Let $U$ be a set of elements, where a fraction $p$ of the elements
    have a certain property $\propertyC$. Let $\Sample$ be a uniform
    random sample from $U$ (with replacement) of size $m$. Let $Y$ be the
    number of elements in $\Sample$ that have the property
    $\propertyC$. We have that
    \begin{equation*}
        \Prob{ \Bigl. \Ex{Y} - t \sqrt{m}/2 \leq Y \leq \Ex{Y} + t \sqrt{m}/2
        } \GEQ 1- \frac{1}{t^2}.
    \end{equation*}
\end{lemma}
\begin{proof}
    Let $Y_i = \propertyX{r_i}$ be an indicator variable that is $1$ if
    the $i$\th sample $r_i$ has the property $\propertyC$, for
    $i=1,\ldots, m$.  Observe that
    \begin{equation*}
        p = \Ex{Y_i} = \frac{\numC}{n}
        \qquad\text{and}\qquad
        \Var{Y_i} = p(1-p).        
    \end{equation*}
    For the random variable $Y = \sum_{i=1}^m Y_i$, by linearity of
    expectation and \Eqref{var-bionomial}, we have $\Ex{Y} = mp$ and
    $\Var{Y} = m p (1-p) \leq m/4$, implying that
    $\sigma_Y = \sqrt{\Var{Y}} \leq \sqrt{m}/2$. Since
    $t \sigma_Y \leq t \sqrt{m}/2$, by \thmrefY{Chebyshev's
       inequality}{Chebyshev:inequality} we have that
    \begin{equation*}
        \Prob{ \cardin{Y - \Ex{Y}} \geq t \sqrt{m}/2
           \Bigr.
        }
        \LEQ
        \Prob{\Bigl. \cardin{Y - \Ex{Y}} \geq t \sigma_Y}
        \LEQ
        \frac{1}{t^2}.
    \end{equation*}
    Taking the complement yields
    \begin{equation*}
        \Prob{ \cardin{Y - \Ex{Y}} \leq t \sqrt{m}/2 } \geq 1 - \frac{1}{t^2},
    \end{equation*}
    which is equivalent to the stated bound.
\end{proof}

\begin{example}[Numerical example---skip if you hate numbers.]
    Consider a setting where $U = \{ 0,1 \}$ and only $1$ has the desired
    property and as such $p=1/2$. Let us sample $m=4\cdot 10^6$ elements---this
    is completely equivalent to flipping a coin $m$ times. The above says
    that the number of $1$s (i.e., heads) we get is concentrated. Indeed,
    $\sigma_Y = \sqrt{m}/2 = 1000$ and $\Ex{Y}=m/2$, and by the \lemrefY{above
       lemma}{Chebyshev-1} that, we have that
    \begin{equation*}
        \Prob{ \cardin{Y - 2\cdot 10^6} \leq t 1000 }
        \geq 1 - \frac{1}{t^2}.
    \end{equation*}
    For $t=10$, this implies that the probability that the total number
    of heads in $m=4\cdot 10^6$ coin flips is in the range
    \begin{equation*}
        [2,000,000 - 10,000, 2,000,000 + 10,000]
    \end{equation*}
    with probability at least $99\%$. That is pretty crazy---flipping a
    coin is a totally random process for each coin flip, but in the
    aggregate it is pretty predictable.
\end{example}

\begin{example}
    If the size of $U$ is known, and you do not know $p$, then a natural
    estimate for $p$ is to compute how many of the $m$ samples have this
    property.  That is, $\widehat{p} = Y/m$ is an estimate of $p$ (i.e.,
    the fraction of the sample that has the desired property). In
    particular, we can now estimate how many elements in $U$ have the
    property---that is $\widehat{p} n = \tfrac{n}{m} Y$.
\end{example}


\begin{lemma}
    \lemlab{estimate:chebyshev}%
    %
    In the setting of \lemref{Chebyshev-1}, let $n= \cardin{U}$ and
    assume that $\numC$ of them have the property (i.e.,
    $p=\tfrac{\numC}{n}$). Let $Z = \tfrac{n}{m}Y$ be the estimate for
    $\numC$.  Then, for any $t\geq 1$, we have that
    \begin{equation*}
        \Prob{ \Bigl. \smash{\numC - t \frac{n}{2\sqrt{m}}%
              \leq%
              Z%
              \leq%
              \numC + t \frac{n}{2\sqrt{m}} } }%
        \geq%
        1- \frac{1}{t^2}.
    \end{equation*}
\end{lemma}
\begin{proof}
    By \lemref{Chebyshev-1}, we have
    \begin{align*}
        \Prob{\bigl. \cardin{Z - \numC } \geq t \frac{n}{2\sqrt{m}} }%
        &=
        \Prob{\bigl. \cardin{Z - \numC } \geq
           \frac{n}{m}t \cdot \frac{\sqrt{m}}{2} }%
        \\&
        \leq
        \Prob{\bigl. \cardin{Z - \numC } \geq
           \frac{n}{m}t \sigma_Y }%
        \\&
        =
        \Prob{\bigl. \cardin{\frac{n}{m}Y - \frac{n}{m} \Ex{Y} } \geq
           \frac{n}{m}t \sigma_Y }%
        \\&%
        =%
        \Prob{\bigl. \cardin{Y - \Ex{Y} } \geq t \sigma_Y }%
        \leq
        \frac{1}{t^2}.
    \end{align*}
    Taking the complement completes the proof.
\end{proof}

\Section{Randomized selection via sampling}


\SubSection{Inverse estimation}

We are given a set $U = \brc{u_1,\ldots, u_n}$ of $n$ distinct numbers. Let
$\EBRY{U}{i}$ denote the $i$\th smallest number in $U$---that is,
$\EBRY{U}{i}$ is the element of \emphi{rank} $i$ in $U$.

\begin{lemma}
    \lemlab{good:interval}%
    %
    Given a set $U$ of $n$ numbers, a rank $k \in \brc{1, \dots, n}$, and
    parameters $t \geq 1$ and $m \geq 1$, one can compute, in
    $O(m \log m)$ time, two numbers $r_{-}, r_+ \in U$, such that:
    \begin{compactenumA}
        \medskip%
        \item The element of rank $k$ in $U$ is in the interval
        $\Interval = [r_-, r_+]$.  \smallskip%
        \smallskip%
        \item There are at most $8 t {n}/ \sqrt{m}$ elements of $U$ in
        $\Interval$.
    \end{compactenumA}
    \medskip%
    The above two properties hold with probability $\geq 1-3/t^2$.
\end{lemma}
\begin{proof}
    \textbf{(A)} Compute a random sample $\Sample$ of $U$ of size $m$
    in $O(m)$ time (assuming the input numbers are given in an array,
    say). Next sort the numbers of $\Sample$ in $O( m \log m)$
    time. Let
    \begin{equation*}
        \ell_{-} = \floor{m \frac{k}{n} - t \sqrt{m}/2 }-1
        \qquad \text{and} \qquad%
        \ell_{+} = \ceil{m \frac{k}{n} + t \sqrt{m}/2 }+1.
    \end{equation*}
    Set $r_{-} = \Sample[ \ell_{-}]$ and $r_+ = \Sample[ \ell_{+}]$ (one needs
    to take care of boundary conditions\footnote{Boundary condition if
       you must: If $\ell_- < 1$, we set $r_- = -\infty$ (or the minimum element
       of $U$); similarly, if $\ell_+ > m$, we set $r_+ = +\infty$ (or the
       maximum element of $U$).}).

    Let $Y$ be the number of elements in the sample $\Sample$ that are
    $\leq \EBRY{U}{k}$.  By \lemref{estimate:chebyshev}, we have
    \begin{equation*}
        \ds%
        \Prob{\Bigl. \Ex{Y} - t \sqrt{m}/2 \leq Y \leq \Ex{Y} + t \sqrt{m}/2
        }
        \GEQ
        1- {1}/{t^2}.
    \end{equation*}
    In particular, if this happens, then since $\Sample$ is sorted and
    $\ell_- < Y < \ell_+$, we have $r_- \leq \EBRY{U}{k} \leq r_+$. %

    \bigskip%
    \noindent
    \textbf{(B)} Let $g= k - t \frac{n}{\sqrt{m}} -3\frac{n}{m}$, and let
    $g^{}_\Sample$ be the number of elements in $\Sample$ that are at
    most $\EBRY{U}{g}$ (i.e., $\leq \EBRY{U}{g}$). Arguing as above, we
    have that
    \begin{equation*}
        \Prob{ g^{}_\Sample \leq \frac{g}{n} m + t \sqrt{m}/2 } \geq
        1- {1}/{t^2}.
    \end{equation*}
    Now
    \begin{align*}
      \frac{g}{n} m + t \sqrt{m}/2 %
      &=
      \frac{m}{n}\pth{
      k - t \frac{n}{\sqrt{m}} -3\frac{n}{m} }  +  t \sqrt{m}/2
      \\&=
      k \frac{m}{n} - t \sqrt{m}  -3 +  t \sqrt{m}/2
      \\&=
      k \frac{m}{n} - t \sqrt{m}/2  -3
      < \ell_{-}.
    \end{align*}
    This implies that the $g$ smallest numbers in $U$ are outside the
    interval $[r_-,r_+]$ with probability $\geq 1-1/t^2$.

    Next, let $h= k + t \frac{n}{\sqrt{m}} +3\frac{n}{m}$. A similar
    argument shows that all the $n-h$ largest elements in $U$ are
    strictly larger than $r_+$ (and hence outside $[r_-, r_+]$).  This
    implies that
    \begin{equation*}
      \cardin{[r_-, r_+] \cap U} \leq h-g = 6 \frac{n}{m} + 2t
      \frac{n}{\sqrt{m}}%
      \leq%
      8 \frac{tn}{\sqrt{m}}.
  \end{equation*}
  In words, since ranks $1,\dots,g$ and $h+1,\dots,n$ are excluded, at
  most $h-g$ candidate elements remain in $\{g+1,\dots,h\}$.
\end{proof}

\subsubsection{Inverse estimation -- intuition}

Here we are trying to give some intuition to the proof of the previous
lemma. Feel free to skip this part if you feel you already understand
what is going on.

\begin{figure}[t]
    \centering
    \IncludeGraphics[width=0.95\linewidth]{\File{figs/inverse_estimation}}
    \caption{Inverse estimation via sampling: Sample pivots $r_-,
      r_+ \in \Sample$ are chosen around the expected rank $\mu$,
      bracketing the rank-$k$ element $s_k$ in the interval $I = [r_-,
      r_+]$ in $U$. See \lemref{good:interval} for details.}
    \figlab{inverse:estimation}
\end{figure}


Given $k$, we are interested in estimating $s_k = \EBRY{U}{k}$
quickly. So, let us take a sample $\Sample$ of size $m$. Let
$\Sample_{\leq s_k}$ be the set of all the numbers in $\Sample$ that
are $\leq s_k$. For $Y = \cardin{\Sample_{\leq s_k}}$, we have that
\begin{math}
  \mu = \Ex{Y} = m \frac{k}{n}.
\end{math}
Furthermore, for any $t\geq 1$, \lemref{estimate:chebyshev} implies
that
\begin{math}
    \ds%
    \Prob{ \mu - t \sqrt{m}/2 \leq Y \leq \mu + t \sqrt{m}/2 } \geq 1-
    {1}/{t^2}.
\end{math}
In particular, with probability $\geq 1- 1/t^2$ the number
$r_{-} =\EBRY{\Sample}{\ell_-}$, for
$\ell_{-} = \floor{\mu - t \sqrt{m}/2 }-1$, is smaller than $s_k$, and
similarly, the number $r_{+} = \EBRY{\Sample}{\ell_{+}}$ of rank
$\ell_{+} = \ceil{\mu + t \sqrt{m}/2 }+1$ in $\Sample$ is larger than
$s_k$.

One can conceptually think about the interval
\begin{math}
    \Interval(k) = [r_-,\, r_+ ]%
\end{math}
as a \emph{confidence interval}---we know that $s_k \in \Interval(k)$
with probability $\geq 1-1/t^2$. But how heavy is this interval?  Namely,
how many elements are there in $\Interval(k) \cap U$?

To this end, consider the interval of ranks, \emph{in the sample},
that might contain the $k$\th element. By the above, this is
\begin{equation*}
  \IntervalA(k, t)%
  =%
  k \frac{m}{n} + \pbrc[]{ -t \sqrt{m}/2 -1, t
  \sqrt{m}/2 +1}.
\end{equation*}
In particular, consider the maximum $\nu\leq k$ such that
$\IntervalA(\nu, t)$ and $\IntervalA(k, t)$ are disjoint. We have the
condition that
\begin{equation*}
    \nu \frac{m}{n}  +t
    \sqrt{m}/2 +1 \leq k \frac{m}{n} -t\sqrt{m}/2 -1%
    %
    \implies%
    %
    \nu  \leq k  - t \frac{n}{\sqrt{m}} -2\frac{n}{m}.
\end{equation*}
Let
\begin{equation*}
    g= k - t \frac{n}{\sqrt{m}} -2\frac{n}{m}
    \qquad\text{and}\qquad
    h = k + t \frac{n}{\sqrt{m}} +2\frac{n}{m}.
\end{equation*}
By construction, the sample rank intervals $\IntervalA(g, t)$,
$\IntervalA(k, t)$, and $\IntervalA(h, t)$ are deterministically pairwise
disjoint.

It is easy to verify (using the same argumentation as above) that with
probability at least $1-3/t^2$, the three confidence intervals
$\Interval(g), \Interval(k)$ and $\Interval(h)$ do not intersect. As
such, we have
\begin{math}
    \cardin{\big.\ts \Interval(k) \cap U\ts}%
    \leq%
    h-g%
    \leq%
    2 t \frac{n}{\sqrt{m}} + 4 \frac{n}{m} \leq 6 t \frac{n}{\sqrt{m}}.
\end{math}

\subsection{Randomized selection}

\subsubsection{The algorithm.}
Given an array $S$ of $n$ numbers and a rank $k$, the algorithm needs to
compute $\EBRY{S}{k}$. To this end, it sets
\begin{equation*}
    t= \ceil{n^{1/8}}
    \qquad\text{ and }\qquad%
    m = \ceil{n^{3/4}}.
\end{equation*}
Using the algorithm of \lemref{good:interval}, in $O( m \log m)$ time,
the algorithm gets two numbers $r_{-}$ and $r_+$, such that
$\EBRY{S}{k} \in [r_{-}, r_{+}]$, and
\begin{equation*}
    \cardin{ \smash{\underbrace{S \cap (r_{-}, r_{+})}_{S_m}} }%
    =%
    O\bigl(tn /\sqrt{m}\,\bigr)%
    =%
    O\pth{ n^{1/8+ 1}  / n^{3/8}}%
    =%
    O(n^{3/4}),
\end{equation*} %
and this happens with good probability (more on that later).
%\bigskip%
%\noindent%
To this end, the algorithm breaks $S$ into three sets:
\begin{compactenumi}
    \smallskip%
    \item $S_< = \Set{ s \in S}{s \leq r_{-}}$,
    \smallskip%
    \item $S_m = \Set{ s \in S}{r_{-} < s < r_{+}}$, and
    \smallskip%
    \item $S_> = \Set{ s \in S}{ r_{+} \leq s}$.
\end{compactenumi}%
% \medskip% This happens with probability $\geq 1 - 3/t^2 \geq 1 - 4/n^{1/4}$.  \snark{$3$ became $4$ because of floor nonsense---you shouldn't care.}
\medskip%
This three-way partition can be done using $2n$ comparisons and in linear
time.  The algorithm computes the rank of $r_-$ in $S$ (it is
$\cardin{S_<}$) and the rank of $r_{+}$ in $S$ (it is
$\cardin{S_<} + \cardin{S_m} + 1$). There are several cases: \smallskip%
\begin{compactitem}
    \item If $\rankY{r_{-}}{ S} > k$ or $\rankY{r_{+}}{S} < k $ then the
    algorithm fails.

    \item If $\cardin{S_m} > 8 t {n}/ \sqrt{m} = \Theta( n^{3 / 4})$,
    then the set $S_m$ is too large, and the algorithm fails.
\end{compactitem}
\medskip%
If either failure occurs, the algorithm restarts from scratch.
\begin{compactitem}
    \smallskip%
    \item If $k = \cardin{S_<}$, the algorithm returns $r_-$ as the
    desired answer.

    \smallskip%
    \item If $k = \cardin{S_<} + \cardin{S_m} + 1$, then the algorithm
    returns $r_+$.

    \smallskip%
    \item Otherwise, $\cardin{S_<} < k \leq \cardin{S_<} + \cardin{S_m}$,
    and the algorithm needs to compute the element of rank
    $\Delta = k - \cardin{S_<}$ in the set $S_m$. This is done by sorting
    $S_m$ and returning the element of rank $\Delta$ in the sorted
    subarray.
    % , and this can be done in
    % $O( \cardin{S_m} \log \cardin{S_m}) = O( n^{3/4} \log n)$ time by
    % using sorting.
\end{compactitem}

\begin{remark}
    Crucially for the analysis, this algorithm is not recursive---it
    performs some rounds of sampling followed by partition, and then a
    single sorting of the interval of interest and extracting the desired
    element from this sorted subset. It is thus fundamentally different
    from \AlgorithmI{QuickSelect}.
\end{remark}

\subsubsection{Analysis.}

For the sake of simplicity of exposition, we ignore any floors/ceilings
in the calculations here\footnote{It is easy to verify the floor function
   only introduces tedium and anxiety in carrying out (essentially) the
   same calculations.}.  The correctness is intuitively easy---the
algorithm clearly returns the desired element (the analysis below shows
that the probability of running an infinite number of rounds is zero,
which implies that the algorithm terminates almost surely\footnote{In
   probability, \emphi{almost surely} means it happens with probability
   $1$, so there is really no almost about it. It is probability's way of
   politely stating there is no escape---for example, almost surely we are
   all going to regret writing/reading this footnote.}).


\paragraph*{Running time.}
By \lemref{good:interval}, with probability $\geq 1-3/n^{1 /4}$, the
algorithm succeeds on the first try.  The running time per round is
$O(n + |S_m| \log |S_m|) = O(n)$, as $\cardin{S_m} = O(n^{3/4})$.




\subparagraph*{Not too many rounds.}

More generally, the probability that the algorithm fails in the first
$\alpha$ tries to get a good interval $[r_{-}, r_{+}]$ is at most
$(3/n^{1/4})^\alpha = O(3^\alpha /n^{\alpha /4})$. Let $Y$ be the number
of rounds the algorithm performs. The running time of the algorithm is
$ O(Yn)$.  The variable $Y$ is a geometric variable and has
\begin{equation}
    \mu = \Ex{Y} \leq  \frac{1}{  1- 3/n^{1/4}} \leq 1 + 4/n^{1/4},
    \eqlab{q-s-few-rounds}
\end{equation}
for $n$ sufficiently large, since $Y$ is a geometric random variable with
success probability
\begin{equation}
    p \geq 1 - 3/n^{1/4}.
    \eqlab{s-p-r-selection}
\end{equation}
Thus, the expected running time of the algorithm is $O(n)$.

\subsubsection{Doing better.}

One can slightly improve the number of comparisons performed by the
algorithm using the following modifications.
\begin{lemma}
    Given the numbers $r_{-}, r_{+}$ computed by \lemref{good:interval},
    one can compute the sets $S_<$, $S_m$, and $S_>$ using in expectation
    (strikingly only) $1.5n + |S_m|/2$ comparisons.
\end{lemma}
\begin{proof}
    We need to compute the sets $S_{<}, S_m, S_>$. Namely, we need to
    compare all the numbers of $S$ to $r_{-}$ and $r_{+}$. Since only
    $O(n^{3/4})$ numbers fall in $S_m$, almost all of the numbers are in
    either $S_<$ or $S_>$. If a number is in $S_<$ (resp. $S_>$), then
    comparing it to $r_-$ (resp. $r_+$) is enough to verify that this is
    indeed the case. Otherwise, perform the other comparison and put the
    element in its proper set (in this case we had to perform two
    comparisons to handle the element).

    So let us guess, by a coin flip, for each element of $S$ whether it
    is in $S_<$ or $S_>$. If we are right, then the algorithm would
    require only one comparison to put it into the right set. Otherwise,
    it would need two comparisons. Let $X_s$ be the random variable that
    is the number of comparisons used by this algorithm for an element
    $s \in S$. We have that if $s \in S_< \cup S_>$ then
    $\Ex{X_s} = 1(1/2) + 2(1/2) = 3/2$. If $s \in S_m$ then both
    comparisons will be performed, and thus $\Ex{X_s}=2$ in this case.

    Thus, the expected number of comparisons for all the elements of $S$,
    by linearity of expectation, is
    $\tfrac{3}{2}(n-|S_m|) + 2 |S_m| = (3/2)n + |S_m|/2$.
\end{proof}


\paragraph*{Expected number of comparisons in a round.}
A round is successful with probability
$\geq 1- 3/n^{1/4}$, see \Eqref{s-p-r-selection}. Thus, in expectation the
algorithm performs
\begin{align*}
    \xi
    &\EQ
    p\cdot \Bigl(1.5n + O(n^{3/4})\Bigr) + (1-p)2n
    \\&
    \LEQ
    1.5n +  O(n^{3/4}) + (3/n^{1/4}) \cdot 2n
    \\&
    \EQ
    1.5n + O(n^{3/4})
\end{align*}
comparisons per round. 



\begin{theorem}
    Given an array $S$ with $n$ numbers and a rank $k$, one can compute
    the element of rank $k$ in $S$ in expected linear time.
    Formally, the resulting algorithm performs in expectation
    $1.5 n + O( n^{3/4} \log n)$ comparisons.
\end{theorem}
\begin{proof}
    The intuitive argument is obvious---the algorithm performs $Y$ rounds,
    and each round takes in expectation $\xi$ comparisons. As the
    expectation of $Y$ is $\Ex{Y} = \frac{1}{1-p} = 1 + O(1/n^{3/4})$, it
    follows the expected number of comparisons is $\Ex{Y} \xi$ which is the
    desired quantity as an easy calculation shows. This argument is
    correct, but formalizing it requires some tedious work.

    Let $F$ be the expected number of comparisons performed by the
    algorithm. Consider the first round, with probability $p$ it
    performed $1.5n + O(n^{3/4})$ comparisons, and with probability $1-p$
    it is a lost round, and the algorithm has to restart. In case of such
    a failure it had to perform (at most) $2n$ comparisons in this round,
    and whatever comparisons the algorithm performs later on. However,
    since the process is memoryless, the expected number of remaining
    comparisons the algorithm has to perform is $F$. Namely, we have that
    \begin{equation*}
        F
        \LEQ
        1.5n + O(n^{3/4}) + (1-p)(2n + F)
        \LEQ  1.5n + O(n^{3/4}) +(1-p) F.
    \end{equation*}
    This implies that
    \begin{align*}
        F
        &\LEQ
        \frac{1.5 n + O(n^{3/4}) }{p}
        \LEQ
        \pth{ 1 + \frac{6}{n^{3/4}}}
        \cdot \pth{1.5 n + O(n^{3/4}) }
        \\&%
        \LEQ%
        1.5n + O(n^{3/4}),
    \end{align*}
    using \Eqref{s-p-r-selection} Wald's identity yields
    $\Ex{C} \leq \Ex{Y}(1.5n + O(n^{3/4} \log n)) = 1.5n + 6n^{3/4} +
    O(n^{3/4} \log n) = 1.5n + O(n^{3/4} \log n) = 1.5n + o(n)$.
\end{proof}

The above implicitly proves a special case of the following.

\begin{theorem}[Wald's Equation]
    Let $(X_n)_{n \ge 1}$ be a sequence of independent and identically
    distributed (i.i.d.) real-valued random variables with
    $\mathbb{E}[|X_1|] < \infty$. Let $Y$ be an integer-valued stopping time
    with respect to the filtration generated by $(X_n)_{n \ge 1}$ such that
    $\mathbb{E}[Y] < \infty$.  Then
    \begin{math}
        \Ex{ \sum_{i=1}^{Y} X_i } = \Ex{Y} \,
        \Ex{X_1}.
    \end{math}
\end{theorem}
