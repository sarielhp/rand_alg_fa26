\section{Explicit expander construction}

We state here a few facts about expander graphs without proofs (see
also \thmref{{f:t:algebraic:graph}}).


\begin{defn}
    And $(n,d,c)$-expander is a $d$-regular bipartite graph
    $\Graph=(X,Y,\Edges)$, where $\cardin{X} = \cardin{Y}=n/2$. Here,
    we require that for any $S \subseteq X$, we have
    \begin{equation*}
        \cardin{\Gamma(S)}\geq \pth{ 1 + c \pth{1- \frac{2|S|}{n}}} \cardin{S}.
    \end{equation*}
\end{defn}

\paragraph*{The Margulis-Gabber-Galil expander.}
For a positive $m$, let $n=2m^2$. Each vertex in $X$ and $Y$ above, is
interpreted as a pair $(a,b)$, where
$a, b \in \ZZ_m = \brc{0,\ldots, m-1}$. A vertex $(a,b) \in X$ is
connected to the vertices
\begin{equation*}
    (a,b), \quad%
    (a,a+b), \quad%
    (a,a+b+1), \quad%
    (a+b,b), \quad \text{and} \quad%
    (a+b+1,b), \quad%
\end{equation*}
in $Y$, where the addition is done module $m$. %

\begin{theorem}[\cite{gg-eclss-81}]
    The above graph is $5$ regular, and it is
    $(n,5,(2-\sqrt{3})/4)$-expander.
\end{theorem}

\paragraph*{Specrtral gap and expansion.}

We remind the reader that a $d$-regular graph, then its adjacency
matrix $\Mtrx(\Graph)$ has (as its biggest eigenvalue) the eigenvalue
$\lambda_1 =d$. In particular, let
$\cardin{\lambda_1} \geq \cardin{\lambda_2} \geq \ldots \geq
\cardin{\lambda_n}$ be the eigenvalues of $\Mtrx(\Graph)$. We then
have the following:

\begin{theorem}
    If $\Graph$ is an $(n,d,c)$-expander then $\Mtrx(\Graph)$ has
    \begin{math}
        \ds%
        \cardin{\lambda_2} \leq d - \frac{c^2}{1024+2c^2}.
    \end{math}
\end{theorem}

\begin{theorem}
    If $\Mtrx(\Graph)$ has $\cardin{\lambda_2} \leq d - \eps$, then
    $\Graph$ is $(n,d,c)$-expander with
    \begin{math}
        \ds
        c \geq \frac{2d \eps - \eps^2}{d^2}.
    \end{math}
\end{theorem}

\section{Rapid mixing for expanders}

%We remind the reader of the following definition of expander.

Here is another equivalent definition of an expander.
\begin{defn}
    Let $\Graph=(V,E)$ be an undirected $d$-regular graph.  The graph
    $\Graph$ is a
    \emphic{$(n,d,c)$-expander}{expander!$[n,d,c]$-expander} (or just
    \emphic{$c$-expander}{expander!$c$}), for every set
    $S \subseteq V$ of size at most $\cardin{V}/2$, there are at least
    $c d \cardin{S}$ edges connecting $S$ and
    $\overline{S} = V \setminus S$; that is
    $ e\bigl( S, \overline{S} \, \bigr) \geq c d \cardin{S}$,
\end{defn}


\paragraph*{Guaranteeing aperiodicity.}

Let $\Graph$ be a $(n,d,c)$-expander. We would like to perform a
random walk on $\Graph$. The graph $\Graph$ is connected, but it might
be periodic (i.e., bipartite). To overcome this, consider the random
walk on $\Graph$ that either stay in the current state with
probability $1/2$ or traverse one of the edges. Clearly, the resulting
Markov Chain (\MC) is aperiodic. The resulting \emphi{transition
   matrix} is
\begin{equation*}
    \TMtrx = \Mtrx / 2d + \IMtrx/2,
\end{equation*}
where $\Mtrx$ is the adjacency matrix of $\Graph$ and $\IMtrx$ is the
identity $n\times n$ matrix. Clearly $\TMtrx$ is doubly
stochastic. Furthermore, if $\EV{i}$ is an eigenvalue of $\Mtrx$, with
eigenvector $v_i$, then
\begin{equation*}
    \TMtrx v_i%
    =%
    \frac{1}{2} \pth[]{ \frac{\Mtrx}{d} + \IMtrx} v_i%
    =%
    \frac{1}{2} \pth[]{ \frac{\EV{i}}{d} + 1} v_i.
\end{equation*}
As such, $\bigl(\EV{i}/{d} + 1 \bigr)/2$ is an eigenvalue of
$\TMtrx$. Namely, if there is a spectral gap in the graph
$\Graph$, there would also be a similar spectral gap in the resulting
\MC. This \MC can be generated by adding to each vertex
$d$ self loops, ending up with a
$2d$-regular graph. Clearly, this graph is still an expander if the
original graph is an expander, and the random walk on it is aperiodic.

From this point on, we would just assume our expander is aperiodic.


\subsection{Bounding the mixing time}

For a \MC with $n$ states, we denote by
$\pi = \pth[]{\bigl.\pi_1, \ldots, \pi_n}$ its stationary
distribution. We consider only nicely behave \MC that fall under
\thmref{fundamental}. As such, no state in the \MC has zero stationary
probability.

\begin{defn}
    Let $\DstrbV{t}$ denote the state probability vector of a Markov
    chain defined by a transition matrix $\TMtrx$ at time $t \geq 0$,
    given an initial distribution $\DstrbV{0}$. The \emphi{relative
       pairwise distance} of the Markov chain at time $t$ is
    \begin{equation*}
        \Delta(t)%
        =%
        \max_i \frac{|\DstrbE{t}{i} - \pi_i|}{\pi_i}.
    \end{equation*}
    Namely, if $\Delta(t)$ approaches zero then $\DstrbV{t}$
    approaches $\pi$.
\end{defn}

We remind the reader that we saw a construction of a constant degree
expander with constant expansion. In its transition matrix $\TMtrx$,
we have that $\TEV{1} = 1$, and $-1 \leq \TEV{2} <1$, and furthermore
the \emphi{spectral gap} $\TEV{1} - \TEV{2}$ was a constant (the two
properties are equivalent, but we proved only one direction of this).

We need a slightly stronger property (that does hold for our expander
construction). We have that
\begin{math}
    \TEV{2} \geq \max_{i=2}^n \bigl|\TEV{i}\bigr|.
\end{math}


\begin{theorem}
    Let $\TMtrx$ be the transition matrix of an aperiodic
    $(n,d,c)$-expander (i.e., the transition matrix of the Markov
    chain).  Let $\TEV{i}$ be the $i$\th eigenvalue of $\TMtrx$, for
    $i=1,\ldots, n$, in decreasing absolute value. Then, for any
    initial distribution $\DstrbV{0}$, we have that
    \begin{equation*}
        \Delta(t) \leq n^{3/2} \bigl( \TEV{2} \bigr)^t.
    \end{equation*}
    Since $\TEV{2}$ is a constant smaller than $1$, the distance
    $\Delta(t)$ drops exponentially with $t$.
\end{theorem}
\begin{proof}
    We have that $\DstrbV{t} = \DstrbV{0}\TMtrx^t$.  Let
    $\Basis(\TMtrx)= \permut{v_1,\ldots, v_n}$ denote the orthonormal
    eigenvector basis of $\TMtrx$ (see
    \defrefpage{orthonormal:basis}), and write $\DstrbV{0}=
    \sum_{i=1}^n \alpha_i v_i$. Since $\TEV{1}=1$, we have that
    \begin{equation*}
        \DstrbV{t}%
        =%
        \DstrbV{0}\TMtrx^t%
        =%
        \sum_{i=1}^n \alpha_i \pth[]{\bigl. v_i\TMtrx^{\, t} }%
        =%
        \sum_{i=1}^n \alpha_i \bigl(\ts\TEV{i}\ts\bigr)^t v_i%
        =%
        \alpha_1 v_1 + \sum_{i=2}^n \alpha_i \bigl(\ts \TEV{i} \ts\bigr)^t
        v_i.
    \end{equation*}
    Since
    $v_1 = \bigl(1/\,\mysqrt{n}, \ldots, 1/\,\mysqrt{n}\ts\bigr)$, and
    \begin{math}
        | \TEV{i} | \leq \TEV{2} < 1,
    \end{math}
    for $i >1$, we have that
    \begin{math}
        \lim_{t\rightarrow \infty}\bigl( \ts \TEV{i}\ts \bigr)^t = 0,
    \end{math}
    and thus
    \begin{equation*}
        \pi%
        =%
        \lim_{t\rightarrow \infty} \DstrbV{t}%
        =%
        \alpha_1 v_1 + \sum_{i=2}^n \alpha_i \pth[]{\,
           \lim_{t\rightarrow \infty}
           \bigl( \ts\TEV{i}\ts \bigr)^t \,} v_i%
        =%
        \alpha_1 v_1.
    \end{equation*}   
    Now, since $v_1, \ldots, v_n$ is an orthonormal basis, and
    \begin{math}
        \ds \DstrbV{0} = \sum_{i=1}^n \alpha_i v_i,
    \end{math}
    we have that
    \begin{math}
        \ds%
        \norm{\smash{\DstrbV{0}}}_2%
        =%
        \smash{\sqrt{\sum_{i=1}^n
              \alpha_i^2}}.
    \end{math}
     Thus implies that
    \begin{align*}
      \norm{\DstrbV{t} - \pi}_1%
      &=%
        \norm{\DstrbV{t} - \alpha_1 v_1}_1%
        =%
        \norm{\sum_{i=2}^n \alpha_i \pth[]{\ts\TEV{i}\ts}^t v_i}_1%
        \leq%
        \sqrt{n} \norm{\sum_{i=2}^n \alpha_i \pth[]{\ts \smash{\TEV{i}} \ts
        }^t v_i}_2%
        =%
        \sqrt{n} \sqrt{\sum_{i=2}^n \pth[]{ \alpha_i\, \pth[]{\ts
        \smash{\TEV{i}} \ts }^t}^2}%
      \\&%
      \leq%
      \sqrt{n} \pth[]{\ts\smash{\TEV{2}}\ts}^t \sqrt{\sum_{i=2}^n \pth{
      \alpha_i}^2} \leq%
      \sqrt{n}\, {\ts\TEV{2}\ts}^t \norm{\DstrbV{0}}_2%
      \leq%
      \sqrt{n}\, \pth{\bigl.\smash{\TEV{2}}}^t
      \norm{\bigl.\smash{\DstrbV{0}}}_1%
      =%
      \sqrt{n}\, \pth{\ts\bigl.\smash{\TEV{2}}\ts}^t,
    \end{align*}
    since $\DstrbV{0}$ is a distribution. Now, since $\pi_i = 1/n$, we
    have
    \begin{equation*}
        \Delta(t)%
        =%
        \max_i \frac{\bigl| {\DstrbE{t}{i} -\pi_i} \bigr| }{\pi_i}%
        =%
        \max_i \,n \,{\bigl|{\DstrbE{t}{i} - \pi_i}\bigr|}%
        \leq%
        n \max_i \norm{\smash{\DstrbV{t}} - \pi}_1 \leq n
        \sqrt{n} \pth{\bigl.\smash{\TEV{2}}}^t.
    \end{equation*}
\end{proof}


\section{Probability amplification by random walks %
   on expanders}

We are interested in performing probability amplification for an
algorithm that is a \CClass{\si{BPP}} algorithm (see \defref{b:p:p}).
It would be convenient to work with an algorithm which is already
somewhat amplified. That is, we assume that we are given a
\CClass{\si{BPP}} algorithm \Alg for a language $L$, such that 

\smallskip

\begin{compactenumA}
    \item If $x \in L$ then $\Big.\Prob{\bigl. \Alg(x) \text{
          accepts}} \geq 199/200$.
    
    \smallskip

    \item If $x \notin L$ then
    $\Big. \Prob{\bigl.\Alg(x) \text{ accepts}} \leq 1/200$.
\end{compactenumA}

\medskip

We assume that $\Alg$ requires a random bit string of length $n$. So,
we have a constant degree expander $\Graph$ (say of degree $d$) that
has at least $200 \cdot 2^n$ vertices. In particular, let 
\begin{equation*}
    U = \cardin{V(\Graph)},
\end{equation*}
and since our expander construction grow exponentially in size (but
the base of the exponent is a constant), we have that $U =
O(2^n)$. (Translation: We can not quite get an expander with a
specific number of vertices. Rather, we can guarantee an expander that
has more vertices than we need, but not many more.)

We label the vertices of $\Graph$ with all the binary strings of
length $n$, in a round robin fashion (thus, each binary string of
length $n$ appears either
\begin{math}
    \ceil{ \bigl. \cardin{V(\Graph)} / 2^n }
\end{math}
or
\begin{math}
    \floor{ \bigl. \cardin{V(\Graph)} / 2^n }
\end{math}
times). For a vertex $v \in \VerticesX{\Graph}$, let $\StringX{v}$
denote the binary string associated with $v$.

Consider a string $x$ that we would like to decide if it is in $L$ or
not. We know that at least $99/100 U$ vertices of $\Graph$ are labeled
with ``random'' strings that would yield the right result if we feed
them into \Alg (the constant here deteriorated from $199/200$ to
$99/100$ because the number of times a string appears is not
identically the same for all strings).

\paragraph*{The algorithm.}
We perform a random walk of length $\mu = \alpha \beta k$ on $\Graph$,
where $\alpha$ and $\beta$ are constants to be determined shortly, and
$k$ is a parameter. To this end, we randomly choose a starting vertex
$X_0$ (this would require $n+O(1)$ bits). Every step in the random
walk, would require $O(1)$ random bits, as the expander is a constant
degree expander, and as such overall, this would require $n + O(k)$
random bits.

Now, lets $X_0, X_1, \ldots, X_\mu$ be the resulting random walk. We
compute the result of
\begin{equation*}
    Y_i = \Alg\pth{x, r_i}, %
    \qquad \text{ for } \qquad i=0,\ldots, \altK, \qquad \text{
       and } \altK = \alpha k,
\end{equation*}
where $r_i = \StringX{X_{i\cdot \beta }}$.  Specifically, we use the
strings associated with nodes that are in distance $\beta$ from each
other along the path of the random walk.  We return the majority of
the bits $Y_0, \ldots, Y_{\alpha k}$ as the decision of whether $x \in
L$ or not.

We assume here that we have a \emphi{fully explicit} construction of
an expander. That is, given a vertex of an expander, we can compute
all its neighbors in polynomial time (in the length of the index of
the vertex). While the construction of expander shown is only explicit
it can be made fully explicit with more effort.


\subsection{The analysis}

\paragraph*{Intuition.}

Skipping every $\beta$ nodes in the random walk corresponds to
performing a random walk on the graph $\Graph^\beta$; that is, we
raise the graph to power $\beta$. This new graph is a much better
expander (but the degree had deteriorated). Now, consider a specific
input $x$, and mark the bad vertices for it in the graph
$\Graph$. Clearly, we mark at most $1/100$ fraction of the
vertices. Conceptually, think about these vertices as being uniformly
spread in the graph and far apart. From the execution of the algorithm
to fail, the random walk needs to visit $\alpha k/2$ bad vertices in
the random walk in $\Graph^k$. However, the probability for that is
extremely small - why would the random walk keep stumbling into bad
vertices, when they are so infrequent?



\paragraph*{The real thing.}

Let $\TMtrx$ be the transition matrix of $\Graph$. We assume, as
usual, that the random walk on $\Graph$ is aperiodic (if not, we can
easily fix it using standard tricks), and thus ergodic. Let $\BMtrx =
\TMtrx^\beta$ be the transition matrix of the random walk of the
states we use in the algorithm. Note, that the eigenvalues (except the
first one) of $\BMtrx$ ``shrink''. In particular, by picking $\beta$
to be a sufficiently large constant, we have that 
\begin{equation*}
  \TEV{1}\pth{\bigl. \BMtrx} =1%
  \text{\quad and \quad}%
  \cardin{\smash{\EV{i}\pth{\bigl.\BMtrx}}} \leq \frac{1}{10},%
  \qquad \text{ for }i=2,\ldots, U.
\end{equation*}

For the input string $x$, let $\WMtrx$ be the matrix that has $1$ in
the diagonal entry $\WMtrx_{ii}$, if and only $\Alg(x, \StringX{i})$
returns the right answer, for $i=1,\ldots, U$. (We remind the reader
that $\StringX{i}$ is the string associated with the $i$\th vertex,
and $U =\cardin{\VerticesX{\Graph}}$.)  The matrix $\WMtrx$ is zero
everywhere else. Similarly, let $\NWMtrx =\IMtrx - \WMtrx$ be the
``complement'' matrix having $1$ at $\NWMtrx_{ii}$ iff $\Alg(x,
\StringX{i})$ is incorrect. We know that $\WMtrx$ is a $U \times U$
matrix, that has at least $(99/100)U$ ones on its diagonal.

\begin{lemma}
    \lemlab{eigenvalues:transition}%
    %
    Let $\TMtrx$ be a symmetric transition matrix, then all its
    eigenvalues of $\TMtrx$ are in the range $[-1,1]$.
\end{lemma}

\begin{proof}
    Let $p \in \Re^n$ be an eigenvector with eigenvalue $\lambda$. Let
    $p_i$ be the coordinate with the maximum absolute value in $p$. We
    have that
    \begin{equation*}
        \cardin{\bigl.\lambda p_i}%
        =%
        \cardin{\Bigl.\pth[]{\bigl. p\TMtrx}_i}%
        =%
        \cardin{\sum_{j=1}^U p_j \TMtrx_{ji} }%
        \leq%
        \sum_{j=1}^U \cardin{p_j} \cardin{ \TMtrx_{ji} }%
        \leq%
        \cardin{p_i} \sum_{j=1}^U \cardin{ \TMtrx_{ji} }%
        =%
        \cardin{\bigl. p_i}.
    \end{equation*}
    This implies that $\cardin{\lambda} \leq 1$.
    
    (We used the symmetry of the matrix, in implying that $\TMtrx$
    eigenvalues are all real numbers.)
\end{proof}

\begin{lemma}
    \lemlab{t:matrix:preserves}%
    %
    Let $\TMtrx$ be a symmetric transition matrix, then for any $p \in
    \Re^n$, we have that $\norm{p \TMtrx}_2 \leq \norm{ p }_2$.
\end{lemma}

\begin{proof}
    Let $\Basis(\TMtrx)= \permut{v_1,\ldots, v_n}$ denote the
    orthonormal eigenvector basis of $\TMtrx$, with eigenvalues
    $1=\lambda_1, \ldots, \lambda_n$. Write $p = \sum_i \alpha_i v_i$,
    and observe that
    \begin{equation*}
        \norm{ \bigl. p \TMtrx \ts}_2%
        =%
        \norm{\sum_i \alpha_i v_i \TMtrx \ts}_2%
        =%
        \norm{\sum_i \alpha_i \lambda_i v_i }_2%
        =%
        \sqrt{\sum_i \alpha_i^2 \lambda_i^2}%
        \leq %
        \sqrt{\sum_i \alpha_i^2 }%
        =%
        \norm{\bigl.p \ts}_2,
    \end{equation*}
    since $\cardin{\lambda_i} \leq 1$, for $i=1,\ldots, n$, by
    \lemref{eigenvalues:transition}.
\end{proof}


\begin{lemma}
    \lemlab{shrink:preserve}%
    %
    Let $\BMtrx= \TMtrx^\beta$ be the transition matrix of the graph
    $\Graph^\beta$.  For all vectors $p \in \Re^n$, we have:
    \begin{compactenumi}
        \item
        $\norm{ p \BMtrx \WMtrx} \leq \norm{p}$, and
        \smallskip%
        \item
        $\norm{\bigl. p \smash{\BMtrx \NWMtrx } } \leq \norm{p}/5$.
    \end{compactenumi}
\end{lemma}

\begin{proof}
    (i) Since multiplying a vector by $\WMtrx$ has the effect of
    zeroing out some coordinates, its clear that it can not enlarge
    the norm of a matrix. As such, $\norm{ p \BMtrx \WMtrx}_2 \leq
    \norm{ p \BMtrx}_2 \leq \norm{p}_2$ by
    \lemref{t:matrix:preserves}.

    \smallskip%
    (ii) Write $p = \sum_i \alpha_i v_i$, where $v_1, \ldots, v_n$ is
    the orthonormal basis of $\TMtrx$ (and thus also of $\BMtrx$),
    with eigenvalues $1=\TEV{1}, \ldots, \TEV{n}$. We remind the
    reader that $v_1= (1,1, \ldots,1)/\sqrt{n}$. Since $\NWMtrx$
    zeroes out at least $99/100$ of the entries of a vectors it is
    multiplied by (and copy the rest as they are), we have that
    \begin{equation*}
        \norm{\smash{v_1 \NWMtrx}} \leq \sqrt{ (n/100) (1/\sqrt{n})^2
        } \leq 1/10 = \norm{v_1} /10.        
    \end{equation*}
    Now, for any $x\in \Re^U$, we have
    $\norm{\smash{x \NWMtrx}} \leq \norm{x}$. As such, we have that
    \begin{align*}
      \norm{ p \BMtrx \NWMtrx}_2%
      &=%
        \norm{ \sum_i \alpha_i v_i
        \BMtrx \NWMtrx}_2%
        \leq%
        \norm{ \alpha_1 v_1 \BMtrx \NWMtrx} + \norm{\sum_{i=2}^U
        \alpha_i v_i \BMtrx \NWMtrx}%
        \displaybreak[0]%
      \\%
      & \leq %
        \norm{ \alpha_1 v_1 \NWMtrx} + \Bigl\|  \Bigl( \sum_{i=2}^U
        \alpha_i v_i \TEV{i}^\beta \Bigr) \NWMtrx \,\Bigr\|%
        \leq%
        \frac{\cardin{\alpha_1}}{10} + \norm{{\sum_{i=2}^U \alpha_i
        v_i \TEV{i}^\beta}\;}%
        \displaybreak[0]%
      \\%
      &%
        \leq%
        \frac{\cardin{\alpha_1}}{10} + \sqrt{\sum_{i=2}^U
        \pth{\alpha_i \TEV{i}^\beta}^2}%
        \leq %
        \frac{\cardin{\alpha_1}}{10} + \frac{1}{10} \sqrt{\sum_{i=2}^U
        \alpha_i^2}%
        \leq %
        \frac{\norm{p}}{10} + \frac{1}{10}\norm{p} \leq
        \frac{\norm{p}}{5},
    \end{align*}
    since $\bigl| {\lambda_i^\beta} \bigr| \leq 1/10$, for
    $i=2,\ldots, n$.
\end{proof}


Consider the strings $r_0, \ldots, r_\altK$. For each one of these
strings, we can write down whether its a ``good'' string (i.e., $\Alg$
return the correct result), or a bad string. This results in a binary
pattern $b_0, \ldots, b_k$. Given a distribution $p \in \Re^U$ on the
states of the graph, its natural to ask what is the probability of
being in a ``good'' state. Clearly, this is the quantity $\norm{p
   \WMtrx}_1$. Thus, if we are interested in the probability of a
specific pattern, then we should start with the initial distribution
$p^{0}$, truncate away the coordinates that represent an invalid
state, apply the transition matrix, again truncate away forbidden
coordinates, and repeat in this fashion till we exhaust the
pattern. Clearly, the $\ell_1$-norm of the resulting vector is the
probability of this pattern. To this end, given a pattern $b_0,\ldots,
b_k$, let $\Sequence = \permut{S_0, \ldots, S_\altK}$ denote the
corresponding sequence of ``truncating'' matrices (i.e., $S_i$ is
either $\WMtrx$ or $\NWMtrx$). Formally, we set $S_i=\WMtrx$ if
$\Alg(x, r_i)$ returns the correct answer, and set $S_i = \NWMtrx$
otherwise.

The above argument implies the following lemma.

\begin{lemma}
    For any fixed pattern $b_0, \ldots, b_\altK$ the probability of
    the random walk to generate this pattern of random strings is
    $\displaystyle \norm{\smash{p^{(0)} S_0 \BMtrx S_1 \ldots \BMtrx
          S_\altK}}_1$, where
    $\Sequence = \permut{S_0, \ldots, S_\altK}$ is the sequence of
    $\WMtrx$ and $\NWMtrx$ encoded by this pattern.
\end{lemma}


\begin{theorem}
    The probability that the majority of the outputs $\Alg(x,r_0),
    \Alg(x,r_1), \ldots, \Alg(x,r_k)$ is incorrect is at most $1/2^k$.
\end{theorem}
\begin{proof}
    The majority is wrong, only if (at least) half the elements of the
    sequence $\Sequence = \permut{S_0, \ldots, S_\altK}$ belong to
    $\NWMtrx$. Fix such a ``bad'' sequence $\Sequence$, and observe
    that the distributions we work with are vectors in $\Re^U$. As
    such, if $p^{0}$ is the initial distribution, then we have that
    \begin{equation*}
        \Prob{\bigl. \Sequence}%
        =%
        \norm{p^{(0)} S_0 \BMtrx S_1 \ldots \BMtrx S_\altK}_{\mathbf{1}
           }%
        \leq%
        \sqrt{U} \norm{p^{(0)} S_0 \BMtrx S_1 \ldots \BMtrx
           S_\altK}_{\mathbf{2}} \leq%
        \sqrt{U}\frac{1}{5^{\altK/2}} \norm{p^{(0)}}_2,
    \end{equation*}
    by \lemref{L:1:to:L:2} below (i.e., Cauchy-Schwarz inequality) and
    by repeatedly applying \lemref{shrink:preserve}, since half of the
    sequence $\Sequence$ are $\NWMtrx$, and the rest are $\WMtrx$.
    The distribution $p^{(0)}$ was uniform, which implies that
    $\norm{p^{(0)} }_2 = 1/\sqrt{U}$. As such, let $\mathcal{S}$ be
    the set of all bad patterns (there are $2^{\altK-1}$ such ``bad''
    patterns).  We have
    \begin{equation*}
        \Prob{\Bigl.\, \text{majority is bad}\, }%
        \leq%
        2^\altK \sqrt{U} \frac{1}{5^{\altK/2}} \norm{p^{(0)}}_2%
        =%
        (4/5)^{\altK/2} = (4/5)^{\alpha k/2} \leq \frac{1}{2^k},
    \end{equation*}
    for $\alpha = 7$.
\end{proof}


\subsection{Some standard inequalities}

\begin{lemma}
    \lemlab{L:1:to:L:2}%
    %
    For any vector $\vecA = \pth[]{v_1,\ldots, v_d} \in \Re^d$, we
    have that $\norm{\vecA}_1 \leq \sqrt{d} \norm{\vecA}_2$.
\end{lemma}

\begin{proof}
    We can safely assume all the coordinates of $\vecA$ are positive.
    Now, 
    \begin{equation*}
        \norm{\vecA}_1%
        =%
        \sum_{i=1}^d {v_i}%
        =%
        \sum_{i=1}^d {v_i} \cdot 1 %
        =%
        \cardin{\dotprod{\vecA}{(1,1, \ldots, 1)}} \leq%
        \sqrt{\sum_{i=1}^d v_i^2 } \sqrt{\sum_{i=1}^d 1^2}%'
        =%
        \sqrt{d} \norm{\bigl.\vecA},
    \end{equation*}
    by the Cauchy-Schwarz inequality.
\end{proof}
