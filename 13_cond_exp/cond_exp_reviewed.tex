\Section{Conditional expectation}

\begin{defn}
    For two discrete random variables $X$ and $Y$, and any value $y$
    with $\Prob{Y = y} > 0$, the \emphOnly{conditional expectation} of
    $X$ given $Y = y$ is
    \begin{equation*}
        \ExCond{X}{Y = y}%
        =%
        \sum_{x} x \ProbCond{X = x}{Y = y}.
    \end{equation*}
    \remX{Clarified that $y$ satisfies $\Prob{Y=y} > 0$ to ensure the conditional probability is well-defined, and changed the summation domain from the sample space $\Omega$ to the support of $X$ (consistent with the proofs below).}%
    The expression $\ExCond{X}{Y}$ denotes the
    \emphOnly{conditional expectation}\index{conditional expectation} of $X$
    given $Y$; it is the random variable $g(Y)$, where $g(y) = \ExCond{X}{Y = y}$.
\end{defn}

\chgY{As such, the conditional expectation is a function from the value of
$y$, to the average value of $X$. As such, one can think of
conditional expectation as a function $f(y) = \ExCond{X}{Y=y}$.}{In other words, $g(y) = \ExCond{X}{Y=y}$ is a deterministic function mapping each realized value $y$ of $Y$ to the conditional expectation of $X$. When evaluated on the random variable $Y$ itself, $\ExCond{X}{Y} = g(Y)$ is a random variable that takes the numerical value $g(y)$ whenever $Y = y$.}
\remX{The original text described $\ExCond{X}{Y}$ as a ``shorthand for $\ExCond{X}{Y=y}$'' and repeated ``As such''. In probability theory, $\ExCond{X}{Y=y}$ is a deterministic number (for a given $y$), whereas $\ExCond{X}{Y}$ is a random variable. Making this explicit is essential because Lemma~\ref{lemma:conditional:expectation} takes the expectation of this random variable.}


\begin{fragment}{conditional_expectation}
\begin{lemma}%
    \lemlab{conditional:expectation}%
    % 
    For any two random variables $X$ and $Y$, we have
    \begin{equation*}
        \Ex{ \bigl. \ExCond{X}{ Y} } = \Ex{ \bigl. X}.
    \end{equation*}
\end{lemma}
\end{fragment}
\begin{proof}
    Expanding the expectation of the random variable
    $\ExCond{X}{Y}$ over all values $y$ with $\Prob{Y=y} > 0$:
    \begin{align*}
        \Ex{\bigl. \ExCond{X}{Y}}%
        &=%
        \sum_y \Prob{Y = y} \ExCond{X}{Y = y} \\
        &=%
        \sum_y \Prob{Y = y} \frac{\sum_{x} x \Prob{X = x \cap Y = y}}{\Prob{Y = y}} \\
        &=%
        \sum_{y} \sum_{x} x \Prob{X = x \cap Y = y} \\
        &=%
        \sum_{x} x \sum_y \Prob{X = x \cap Y = y} \\
        &=%
        \sum_{x} x \Prob{X = x}
        =%
        \Ex{X}.
    \end{align*}
    \remX{Unified the inline math and `align*' environment into a single display. In the original text, $\ExExt{Y}{\dots}$ notationally placed the realization index $y$ inside an expectation with respect to $Y$; expanding directly as $\sum_y \Prob{Y=y} \ExCond{X}{Y=y}$ resolves this notation clash.}%
\end{proof}

\begin{fragment}{product_expectation}
\begin{lemma}
    \lemlab{expectations:product}%
    % 
    For any two random variables $X$ and $Y$, we have
    \begin{equation*}
        \Ex{\bigl.Y \cdot \ExCond{X }{Y}} = \Ex{\bigl. X Y}.
    \end{equation*}
\end{lemma}
\end{fragment}
\begin{proof}
    We have that
    \begin{align*}
        \Ex{\bigl. Y \cdot \ExCond{X}{Y}}%
        &=%
        \sum_y \Prob{Y = y} \cdot y \cdot \ExCond{X}{Y = y} \\
        &=%
        \sum_y \Prob{Y = y} \cdot y \cdot \frac{\sum_{x} x \Prob{X = x \cap Y = y}}{\Prob{Y = y}} \\
        &=%
        \sum_x \sum_y x y \cdot \Prob{\bigl. X = x \cap Y = y}
        =%
        \Ex{\bigl. X Y}.
    \end{align*}
    \remX{Changed the inline math in the lemma statement to a displayed `equation*' to fix an overfull `\textbackslash{}hbox' (51pt too wide). In the proof, replaced the split inline-math and single-line `equation*' by an `align*' environment to eliminate a 54.5pt overfull `\textbackslash{}hbox'. Also unified notation to use $\ExCond{X}{Y}$ and $\ExCond{X}{Y=y}$ consistently instead of switching to $\Ex{X \sep{Y}}$.}%
    \remX{Pedagogical note: Lemma~\ref{lemma:expectations:product} is a special case of the general ``taking out what is known'' property of conditional expectation: for any function $g(Y)$, one has $\Ex{g(Y) \ExCond{X}{Y}} = \Ex{g(Y) X}$. Here $g(Y) = Y$.}%
\end{proof}

\SubSection{Concentration from conditional expectation}

\begin{lemma}
    \lemlab{concentration:cond:exp}%
    Let $X_1, \ldots, X_n$ be independent random variables, that with
    equal probability are $0$ or $1$.  We have that
    $\Prob{ \sum_i X_i < n/4} < 0.9^n$ and
    $\Prob{ \sum_i X_i > (3/4) n} < 0.9^n$.
\end{lemma} 
\begin{proof}
    Let $Y_0 = 1$. For $i = 1, \ldots, n$, if $X_i = 1$, then we set
    $Y_i = Y_{i-1}$, and if $X_i = 0$, then we set $Y_i = Y_{i-1}/2$.
    Equivalently, $Y_i = Y_{i-1} (1/2)^{1 - X_i}$.
    Since $X_i$ is independent of $X_1, \ldots, X_{i-1}$ (and thus
    independent of $Y_{i-1}$), we have
    \begin{equation*}
        \ExCond{Y_i}{Y_{i-1}}%
        =%
        \frac{1}{2} \frac{Y_{i-1}}{2} +
        \frac{1}{2} Y_{i-1}
        = \frac{3}{4} Y_{i-1}.
    \end{equation*}
    \remX{Clarified that $X_i$ is independent of $Y_{i-1}$ because $Y_{i-1}$ depends only on the previous variables $X_1, \ldots, X_{i-1}$.}%
    Therefore, by \lemref{conditional:expectation} we have
    \begin{equation*}
        \Ex{Y_i}%
        =%
        \Ex{ \bigl. \ExCond{Y_i}{Y_{i-1}}}
        =%
        \Ex{ \frac{3}{4} Y_{i-1}}%
        =%
        \frac{3}{4}\Ex{  Y_{i-1}}%
        =%
        \pth{\frac{3}{4}}^i.
    \end{equation*}
    In particular, $\Ex{Y_n} = (3/4)^n$.
    Now, if $\sum_i X_i > (3/4) n$, then the number of zeros is
    $n - \sum_i X_i < n/4$, which implies 
    \begin{equation*}
        Y_n = (1/2)^{n - \sum_i X_i} > (1/2)^{n/4}.
    \end{equation*}
    We are now ready for our conclusions:
    \begin{align*}
        \Prob{ \sum\nolimits_i X_i > (3/4) n }
        &\leq%
        \Prob{ Y_n \geq (1/2)^{n/4}}
        \leq%
        \frac{\Ex{Y_n}}{(1/2)^{n/4}} \\
        &=%
        \frac{(3/4)^n}{(1/2)^{n/4}}
        =%
        \pth{\frac{3 \cdot 2^{1/4}}{4}}^n
        \leq
        0.9^n,
    \end{align*}
    \remX{Corrected the first relation from `$=$' to `$\leq$': since $Y_n = (1/2)^{n - \sum X_i}$, the event $Y_n \geq (1/2)^{n/4}$ corresponds to $\sum X_i \geq (3/4)n$, which strictly contains $\sum X_i > (3/4)n$ whenever $(3/4)n$ is an integer. Also corrected the subsequent `$\leq$' to `$=$' when substituting $\Ex{Y_n} = (3/4)^n$, and split the display into an `align*' environment to eliminate a 59.7pt overfull `\textbackslash{}hbox'.}%
    where the second inequality is Markov's inequality, and the final
    inequality holds because $3 \cdot 2^{1/4} / 4 \approx 0.8919 < 0.9$.
    By symmetry (considering $1 - X_i$), we also have
    \begin{equation*}
        \Prob{ \sum\nolimits_i X_i < (1/4) n }
        =%
        \Prob{ \sum\nolimits_i X_i > (3/4) n }
        <%
        0.9^n.
    \end{equation*}
    \remX{Placed the symmetric lower-tail bound in a display to eliminate a 2.7pt inline overfull `\textbackslash{}hbox'.}%
\end{proof}
