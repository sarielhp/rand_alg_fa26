\Section{Conditional expectation}

\begin{defn}
    For two discrete random variables $X$ and $Y$, and any value $y$
    with $\Prob{Y = y} > 0$, the \emphOnly{conditional expectation} of
    $X$ given $Y = y$ is
    \begin{equation*}
        \ExCond{X}{Y = y}%
        =%
        \sum_{x} x \ProbCond{X = x}{Y = y}.
    \end{equation*}
    The expression $\ExCond{X}{Y}$ denotes the \emphOnly{conditional
       expectation}\index{conditional expectation} of $X$ given $Y$; it
    is the random variable $g(Y)$, where $g(y) = \ExCond{X}{Y = y}$.
\end{defn}

In other words, $g(y) = \ExCond{X}{Y=y}$ is a deterministic function
mapping each realized value $y$ of $Y$ to the conditional expectation of
$X$. When evaluated on the random variable $Y$ itself,
$\ExCond{X}{Y} = g(Y)$ is a random variable that takes the numerical
value $g(y)$ whenever $Y = y$.

\begin{fragment}{conditional_expectation}
\begin{lemma}%
    \lemlab{conditional:expectation}%
    % 
    For any two random variables $X$ and $Y$, we have
    \begin{equation*}
        \Ex{ \bigl. \ExCond{X}{ Y} } = \Ex{ \bigl. X}.
    \end{equation*}
\end{lemma}
\end{fragment}
\begin{proof}
    Expanding the expectation of the random variable
    $\ExCond{X}{Y}$ over all values $y$ with $\Prob{Y=y} > 0$:
    \begin{align*}
        \Ex{\bigl. \ExCond{X}{Y}}%
        &=%
        \sum_y \Prob{Y = y} \ExCond{X}{Y = y} \\
        &=%
        \sum_y \Prob{Y = y} \frac{\sum_{x} x \Prob{X = x \cap Y = y}}{\Prob{Y = y}} \\
        &=%
        \sum_{y} \sum_{x} x \Prob{X = x \cap Y = y} \\
        &=%
        \sum_{x} x \sum_y \Prob{X = x \cap Y = y} \\
        &=%
        \sum_{x} x \Prob{X = x}
        =%
        \Ex{X}.
    \end{align*}
\end{proof}

\begin{fragment}{product_expectation}
\begin{lemma}
    \lemlab{expectations:product}%
    % 
    For any two random variables $X$ and $Y$, we have
    \begin{equation*}
        \Ex{\bigl.Y \cdot \ExCond{X }{Y}} = \Ex{\bigl. X Y}.
    \end{equation*}
\end{lemma}
\end{fragment}
\begin{proof}
    We have that
    \begin{align*}
        \Ex{\bigl. Y \cdot \ExCond{X}{Y}}%
        &=%
        \sum_y \Prob{Y = y} \cdot y \cdot \ExCond{X}{Y = y} \\
        &=%
        \sum_y \Prob{Y = y} \cdot y \cdot \frac{\sum_{x} x \Prob{X = x \cap Y = y}}{\Prob{Y = y}} \\
        &=%
        \sum_x \sum_y x y \cdot \Prob{\bigl. X = x \cap Y = y}
        =%
        \Ex{\bigl. X Y}.
    \end{align*}
\end{proof}

\SubSection{Concentration from conditional expectation}

\begin{lemma}
    \lemlab{concentration:cond:exp}%
    Let $X_1, \ldots, X_n$ be independent random variables, that with
    equal probability are $0$ or $1$.  We have that
    $\Prob{ \sum_i X_i < n/4} < 0.9^n$ and
    $\Prob{ \sum_i X_i > (3/4) n} < 0.9^n$.
\end{lemma} 
\begin{proof}
    Let $Y_0 = 1$. For $i = 1, \ldots, n$, if $X_i = 1$, then we set
    $Y_i = Y_{i-1}$, and if $X_i = 0$, then we set $Y_i = Y_{i-1}/2$.
    Equivalently, $Y_i = Y_{i-1} (1/2)^{1 - X_i}$.
    Since $X_i$ is independent of $X_1, \ldots, X_{i-1}$ (and thus
    independent of $Y_{i-1}$), we have
    \begin{equation*}
        \ExCond{Y_i}{Y_{i-1}}%
        =%
        \frac{1}{2} \frac{Y_{i-1}}{2} +
        \frac{1}{2} Y_{i-1}
        = \frac{3}{4} Y_{i-1}.
    \end{equation*}
    Therefore, by \lemref{conditional:expectation} we have
    \begin{equation*}
        \Ex{Y_i}%
        =%
        \Ex{ \bigl. \ExCond{Y_i}{Y_{i-1}}}
        =%
        \Ex{ \frac{3}{4} Y_{i-1}}%
        =%
        \frac{3}{4}\Ex{  Y_{i-1}}%
        =%
        \pth{\frac{3}{4}}^i.
    \end{equation*}
    In particular, $\Ex{Y_n} = (3/4)^n$.
    Now, if $\sum_i X_i > (3/4) n$, then the number of zeros is
    $n - \sum_i X_i < n/4$, which implies 
    \begin{equation*}
        Y_n = (1/2)^{n - \sum_i X_i} > (1/2)^{n/4}.
    \end{equation*}
    We are now ready for our conclusions:
    \begin{align*}
        \Prob{ \sum\nolimits_i X_i > (3/4) n }
        &\leq%
        \Prob{ Y_n \geq (1/2)^{n/4}}
        \leq%
        \frac{\Ex{Y_n}}{(1/2)^{n/4}} \\
        &=%
        \frac{(3/4)^n}{(1/2)^{n/4}}
        =%
        \pth{\frac{3 \cdot 2^{1/4}}{4}}^n
        \leq
        0.9^n,
    \end{align*}
    where the second inequality is Markov's inequality, and the final
    inequality holds because $3 \cdot 2^{1/4} / 4 \approx 0.8919 < 0.9$.  By
    symmetry (considering $1 - X_i$), we also have
    \begin{equation*}
        \Prob{ \sum\nolimits_i X_i < (1/4) n }
        =%
        \Prob{ \sum\nolimits_i X_i > (3/4) n }
        <%
        0.9^n.
    \end{equation*}
\end{proof}
