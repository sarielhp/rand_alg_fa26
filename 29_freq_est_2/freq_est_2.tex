\section{Counting number of distinct elements}

We first analyze an idealized estimator based on fully random hash
values. We then turn to pairwise independent sampling.

\subsection{First order statistic}

Let $X_1, \ldots, X_u$ be $u$ independent random variables uniformly
distributed in $[0,1]$. Let $Y = \min (X_1, \ldots, X_u)$. The value $Y$ is
the \emphOnly{first order statistic}\index{first order statistic} of
$X_1,\ldots, X_u$.


%XXX
For a continuous variable $X$, the \emphOnly{probability density
   function}\index{probability density function} (i.e.,
\emphOnly{\pdf}\index{\pdf}) is the ``probability'' of $X$ having this
value. Since this is not well defined, one looks on the
\emphOnly{cumulative distribution function}\index{cumulative distribution
   function} $F(x) = \Prob{X \leq x}$. The \pdf is then the derivative of
the \cdf.  Discrete distribution do not have ordinary \pdf{}s. This can
be simulated by Dirac's $\delta$-functions, but fortunately, this pile of
amusement and headache is outside our scope. In particular, the
probability $\Prob{X=\alpha}$, for any specific $\alpha$, is zero.

\begin{lemma}
    The \emph{probability density function} of $Y$ is
    \begin{equation*}
        f(x) = \binom{u}{1} 1 (1-x)^{u-1}.
    \end{equation*}
\end{lemma}
\begin{proof}
    The minimum exceed $x$ only if all the $u$ numbers are larger than
    $x$. Thus,
    $ F(x)=\Prob{Y \leq x } =1 - (1-x)^u$. As such
    $f(x) = F'(x) = u(1-x)^{u-1}$.
\end{proof}
\begin{proof}[Second version: no need to read it.]
    Considering the \pdf of $X_1$ being $x$, and all other $X_i$s
    being bigger. We have that this \pdf is
    \begin{align*}
        g(x)
        &=%
        \Prob{ (X_1 = x) \cap \smash{\bigcap_{i=2}^u} (X_i>X_1)\Bigr.}
        \\&=
        \ProbCond{\Bigl. \smash{\bigcap_{i=2}^u} (X_i>X_1)}{X_1=x} \Prob{X_1=x}
        \\&=%
        (1-x)^{u-1}.
    \end{align*}
    Since every one of the $X_i$ has equal probability to realize $Y$,
    we have $f(x) = u g(x)$.
\end{proof}

\begin{lemma}
    \lemlab{order:stat}%
    %
    We have $\Ex{Y} = \frac{1}{u+1}$,
    $\Ex{Y^2} = \frac{2}{(u+1)(u+2)}$, and
    $\Var{Y}=\frac{u}{(u+1)^2 (u+2)}$.
\end{lemma}
\begin{proof}
    Let $A,B$ be two independent uniform samples from $[0,1]$,
    independent of $X_1,\ldots,X_u$. Conditioned on $Y$, the probability
    that $A<Y$ is $Y$. On the other hand, $A<Y$ exactly when $A$ is
    the smallest of $u+1$ samples. Thus, by symmetry,
    \[
        \Ex{Y} = \Prob{A<Y} = \frac{1}{u+1}.
    \]
    Similarly, conditioned on $Y$, the probability that both $A$ and
    $B$ are smaller than $Y$ is $Y^2$. This happens exactly when
    $A,B$ are the two smallest of $u+2$ samples. All pairs are equally
    likely to be the two smallest, so
    \[
        \Ex{Y^2} = \Prob{A<Y,\ B<Y}
        = \frac{1}{\binom{u+2}{2}}
        = \frac{2}{(u+1)(u+2)}.
    \]
    Consequently,
    % BEGIN EDIT W5: Split the variance calculation to fit the line.
    \begingroup\color{blue}
    \begin{align*}
        \Var{Y} &= \Ex{Y^2} - \pth{\Ex{Y}}^2\\
        &= \frac{2}{(u+1)(u+2)} - \frac{1}{(u+1)^2}
        = \frac{u}{(u+1)^2(u+2)}.
    \end{align*}
    \endgroup
    % END EDIT W5
\end{proof}

\subsection{The algorithm}

\paragraph*{A single estimator.}



Assume that we have a perfectly random hash function $h$ that randomly
maps $N = \brc{1,\ldots, n}$ to $[0,1]$. Assume that the stream
$\STRM=\strm_1,\ldots,\strm_m$, has $u$ unique numbers in $N$. Then the set
$\{h(\strm_1),\ldots, h(\strm_m)\}$ contains $u$ random numbers uniformly
distributed in $[0,1]$.  The algorithm computes $Y = \min_i h(\strm_i)$.

\paragraph*{Explanation.}
Note that $Y$ is \emph{not} an estimator for $u$ -- instead, as
$\Ex{Y} = 1/(u+1)$, we are estimating $1/(u+1)$. The key observation is
that an $1 \pm \eps$ estimator for $1/(u+1)$, is $1 \pm O(\eps)$ estimator for
$u+1$, which is in turn an $1 \pm O(\eps)$ estimator for $u$.

\begin{lemma}
    Let $\eps, \BadProb \in (0,1)$ be parameters. Given a stream
    $\STRM$ of items from $\{1,\ldots, n\}$ one can return an estimate
    $X$, such that
    \begin{equation}
        \Prob{ (1-\eps/4)\frac{1}{u+1} \leq X \leq
           (1+\eps/4)\frac{1}{u+1}} \geq 1-\BadProb,
        \eqlab{range-est}
    \end{equation}
    where $u$ is the
    number of unique elements in $\STRM$. This requires
    $O\pth{ \frac{1 }{\eps^2 } \log \frac{1}{\BadProb}}$ space.
\end{lemma}
\begin{proof}
    The basic estimator $Y$ has $\mu = \Ex{Y} = \frac{1}{u+1}$ and
    $\nu = \Var{Y}=\frac{u}{(u+1)^2 (u+2)}$. We now plug this
    estimator into the mean/median framework. By \lemref{order:stat},
    for $c$ some absolute constant, this requires maintaining $M$
    estimators, where $M$ is larger than
    \begin{equation*}
        c \frac{4\cdot 16 \nu}{\eps^2 \mu^2 } \log \frac{1}{\BadProb}
        = 
        O\pth{  \frac{u^2 }{\eps^2 u^2 } \log \frac{1}{\BadProb}}
        = 
        O\pth{  \frac{1 }{\eps^2  } \log \frac{1}{\BadProb}}.
    \end{equation*}
\end{proof}

Assume $X$ is in the desired range, and write it in the form
$X = (1+\eta)/(u+1)$, where $|\eta|\leq\eps/4$. For a nonempty stream, $u\geq1$, so
\begin{align*}
    \frac{|(1/X-1)-u|}{u}
    &= \frac{u+1}{u}\frac{|\eta|}{|1+\eta|}
    \leq \frac{2|\eta|}{1-|\eta|}
    \leq \frac{2\eps}{4-\eps}<\eps.
\end{align*}
Thus, $1/X-1$ is a $(1\pm\eps)$-estimate of the number of distinct
elements.

\paragraph*{The algorithm revisited.}
Compute $X$ as above, and output the quantity $1/X-1$.

\bigskip

This immediately implies the following.

\begin{lemma}
    Let $\eps,\BadProb\in(0,1)$. Under the unreasonable\footnote{Criminal
       even!} assumption that we can sample perfectly random functions
    from $\{1,\ldots,n\}$ to $[0,1]$, and store each function and each hash
    value in $O(1)$ words, one can process a nonempty stream with $u$
    distinct elements and return an estimate $\rho$ satisfying
    \[
        \Prob{(1-\eps)u\leq\rho\leq(1+\eps)u}\geq1-\BadProb,
    \]
    using $O(\eps^{-2}\log(2/\BadProb))$ words.
\end{lemma}

\section{Sampling with pairwise independent randomness}
\endgroup
% END EDIT W5

Assume that we have a stream of elements
$\STRM = \strm_1, \ldots, \strm_m$, all taken from the set
$\{1,\ldots, n\}$.  Let $\distinctX{\STRM}$ denote the set of values that
appear in $\STRM$, and set $S=\distinctX{\STRM}$. Thus,
\begin{equation*}
    \Frq_0 = \Frq_0(\STRM) = \cardin{S}
\end{equation*}
is the number of distinct values in the stream $\STRM$.


Assume that we have a random sequence of bits
$\mathcal{B} \equiv B_1, \ldots, B_n$, such that $\Prob{B_i=1} =p$, for some
$p$. Furthermore, we can compute $B_i$ efficiently. Assume that the bits
of $\mathcal{B}$ are pairwise independent, and can be computed using
constant space.


\paragraph*{The sampling algorithm.}
When the $i$\th item $\strm_i$ arrives, we compute $B_{\strm_i}$. If this
bit is $1$, then we insert $\strm_i$ into the random sample $R$ (if it is
already in $R$, there is no need to store a second copy, naturally).


This defines a natural random sample
\begin{equation*}
    R = \Set{i}{B_i =1 \text{ and } i \in S} \subseteq S.
\end{equation*}

\begin{lemma}
    \lemlab{f:e:2:ev}%
    %
    For the above random sample $R$, let $X = \cardin{R}$. We have
    that $\Ex{X} = p \nu$ and $\Var{X} = p \nu - p^2 \nu$, where
    $\nu = \Frq_0(\STRM)$ is the number of distinct elements in $S$.
\end{lemma}
\begin{proof}
    Since $X=\sum_{i\in S}B_i$, linearity of expectation gives
    \[
        \Ex{X}=\sum_{i\in S}\Ex{B_i}=p\nu.
    \]
    Pairwise independence makes the covariances zero, so
    \[
        \Var{X}=\sum_{i\in S}\Var{B_i}
        =\nu p(1-p)=p\nu-p^2\nu.
    \]
\end{proof}
  
\begin{lemma}
    \lemlab{estimator}%
    %
    Let $\eps \in (0,1/4)$.  Given $O(1/\eps^2)$ space, and a
    parameter $N$. Consider the task of estimating the size of
    $\Frq_0 = \cardin{\distinctX{\STRM}}$, where $\Frq_0 > N/4$.
    Then, the algorithm described below outputs one of the following:
    \begin{compactenumA}
        \smallskip%
        \item $\Frq_0 > 2N$.
        \smallskip%
        \item Output a number $\rho$ such that
        $(1-\eps)\Frq_0 \leq \rho \leq (1+\eps)\Frq_0$.
    \end{compactenumA}
    \smallskip%
    (Note, that the two options are not disjoint.)  The output of this
    algorithm is correct, with probability $\geq 7/8$.
\end{lemma}
\begin{proof}
    We set $p= \frac{c}{N\eps^2}$, where $c$ is a constant to be determined
    shortly\footnote{If $p>1$ call the probability police and set its
       value to $1$, which makes what follows uninteresting.}. Let
    $T = p N = O(1/\eps^2)$. We sample a random sample $R$ from $S$, by
    scanning the elements of $S$, and adding $i\in S$ to $R$ if $B_i=1$, If
    the random sample is larger than $8T$, at any point, then the
    algorithm outputs that $\cardin{S} > 2N$.

    In all other cases, the algorithm outputs $\cardin{R} /p$ as the
    estimate for the size of $S$, together with $R$.

    To bound the failure probability, consider first the case that
    $N/4 < \cardin{\distinctX{\STRM}}$. In this case, we have by the
    above, that
    \begin{align*}
        \Prob{\cardin{X - \Ex{X}} > \eps \Ex{X}}%
        &\leq%
        \Prob{\cardin{X - \Ex{X}} > \eps
           \frac{\Ex{X}}{\sqrt{\Var{X}}}
           \sqrt{\Var{X}}}%
        \\&\leq%
        \frac{\Var{X}}{\eps^2(\Ex{X})^2}
        \leq
        \frac{1}{8},
    \end{align*}

    if $\frac{\Var{X}}{\eps^2 (\Ex{X})^2} \leq \frac{1}{8}$, For
    $\nu = \Frq_0 \geq N/4$, this happens if
    $\frac{p \nu}{\eps^2 p^2 \nu^2} \leq \frac{1}{8}$. This in turn is
    equivalent to $8/\eps^2 \leq p \nu $. This in turn happens if
    \begin{equation*}
        \frac{c}{N\eps^2} \cdot \frac{N}{4} \geq \frac{8}{\eps^2},
    \end{equation*}
    which implies that this holds for $c=32$.  Namely, the algorithm
    in this case would output a $(1\pm\eps)$-estimate for
    $\cardin{S}$.

    Thus, assume that $X$ is within the ``good'' range, which happens
    with probability $\geq 7/8$. If the sample gets bigger than $8T$, then
    the above readily implies that the size of $S$ is at least
    $(1-\eps)8T/p > 2N$. Namely, the output of the algorithm is correct in
    this case.
\end{proof}
  
\begin{lemma}
    \lemlab{f:e:0:resolution}%
    %
    Let $\eps \in (0,1/4)$ and $\BadProb \in (0,1)$.  Given
    $O(\eps^{-2} \log \BadProb^{-1})$ space, and a parameter $N$, and
    the task is to estimate $\Frq_0$ of $\STRM$, given that
    $\Frq_0 > N/4$.  Then, there is an algorithm that would output one
    of the following:
    \begin{compactenumA}
        \smallskip%
        \item $\Frq_0 > 2N$.
        \smallskip%
        \item Output a number $\rho$ such that
        $(1-\eps)\Frq_0 \leq \rho \leq (1+\eps)\Frq_0$.
    \end{compactenumA}
    \smallskip%
    (Note, that the two options are not disjoint.)  The output of this
    algorithm is correct, with probability $\geq 1- \BadProb$.
\end{lemma}

\begin{proof}
    We run $O( \log \BadProb^{-1})$ independent copies of
    \lemref{estimator}. If half of them returns that $\Frq_0 > 2N$, then
    the algorithm returns that $\Frq_0 > 2N$. Otherwise, the algorithm
    returns the median of the estimates returned, and return it as the
    desired estimate. The correctness readily follows by a repeated
    application of Chernoff's inequality. Indeed, for the result to be
    wrong, at least a quarter of the estimators have to fail, but the
    probability of failure per estimator is at most $1/8$.
\end{proof}


\begin{lemma}
    Let $\eps \in (0,1/4)$.  Given $O(\eps^{-2} \log^2 n)$ space, one
    can read the stream $\STRM$ once, and output a number $\rho$, such
    that $(1-\eps)\Frq_0 \leq \rho \leq (1+\eps)\Frq_0$. The estimate
    is correct with high probability (i.e., $\geq 1-1/n^{O(1)}$).
\end{lemma}
\begin{proof}
    Let $N_i = 2^i$, for $i=1, \ldots, M$, where $M = \ceil{\lg n}$. Run
    $M$ copies of \lemref{f:e:0:resolution}, for each value of $N_i$,
    with $\BadProb = 1/n^{O(1)}$. Let $Y_1, \ldots, Y_M$ be the outputs of
    these algorithms for the stream. A prefix of these outputs, are going
    to be ``$\Frq_0 > 2N_i$'', Let $j$ be the first $Y_j$ that is a
    number.

  
    % Return this number as the desired estimate. The correctness
    % is easy---the first estimate that is a number, is a correct estimate
    % with high probability. Since $N_M \geq n$, it also follows that $Y_M$
    % must be a number. As such, there is a first number in the sequence,
    % and the algorithm would output an estimate.

    More precisely, there is a minimal index $i$, such that
    $N_i/4 \leq \Frq_0 \leq 2N_i$, and $Y_i$ is a good estimate, with high
    probability. The algorithm returns $Y_i$ as the desired estimate.
    Importantly, all earlier estimates are correct with high probability
    as well.
    % If any of the $Y_j$, for $j < i$, is an estimate, then
    % it is correct (again) with high probability.
\end{proof}
